\chapter*{摘要}
\addcontentsline{toc}{chapter}{摘要}
本文从固定 prefix 上的 KL 镜像子问题出发，统一写出 PMD-Part、PMD-Mean、BPO 与 MDPO 的优化目标、精确最优解和参数梯度。主线结论是：PMD-Part 的闭式 target 是 Gibbs 分布；PMD-Mean 是带单纯形约束的 Lambert-$W$ 投影；MDPO 直接优化 KL 正则化目标，梯度中保留 action importance ratio；BPO 从平方 residual 回归出发；本稿保留完整 reverse-KL 导数并分析其中心化 score 梯度，不把后续线性化、二元 KL 或工程权重作为理论前提。四者可以共享同一个 token-level 最优策略，但离开最优点后的梯度并不相同。重要性采样降低测度偏差却引入 ratio 方差，回归型方法则承担访问测度下的投影偏差。最后核对 OpenKimi、经典 MDPO 和 BPO 论文源码，区分固定 inner budget 与真正的子问题收敛。

\chapter{统一的 token-level 镜像子问题}
\label{ch:unified-problem}

\section{Prefix、行为策略与局部 reward}
\label{sec:notation}

固定一个 prefix $s=(x,a_{<t})$，词表动作为 $a$，行为策略为 $\beh(a\mid s)$，待更新学生策略为 $\cand_\theta(a\mid s)$。以下所有结论都在
\[
\cand_\theta(\cdot\mid s)\ll\beh(\cdot\mid s)
\]
的 support 条件下成立。记局部监督信号为 $f(s,a)$；它可以是即时 per-token reward、优势 $A^\beh(s,a)$，也可以是包含未来 reward 的 soft $Q$。

设 $d_\beh(s)$ 是行为轨迹访问 prefix 的测度。若一个条件 target 在每个 prefix 上可以独立实现，则 $d_\beh(s)>0$ 只决定该 prefix 在总体训练损失中的权重；如果模型共享参数、数据有限或存在未覆盖 prefix，它会改变投影点和估计方差。

\section{KL 镜像最优解}
\label{sec:token-optimum}

给定 Natural Gradient 学习率 $\eta>0$，token-level PMD 子问题为
\begin{equation}
\label{eq:token-subproblem}
 \cand^+(\cdot\mid s)=\argmax_{p\ll\beh(\cdot\mid s)}
 \left\{\E_{a\sim p}[f(s,a)]-\frac{1}{\eta}\KL(p\Vert\beh(\cdot\mid s))\right\}.
\end{equation}
令
\[
 Z_f(s)=\E_{a\sim\beh(\cdot\mid s)}\exp\left(\eta f(s,a)\right).
\]

\begin{theorem}[Token-level Gibbs 解]
\label{thm:gibbs}
若 $0<Z_f(s)<\infty$，式 \eqref{eq:token-subproblem} 的唯一解为
\begin{equation}
\label{eq:gibbs}
 \cand^+(a\mid s)=\frac{\beh(a\mid s)\exp(\eta f(s,a))}{Z_f(s)}.
\end{equation}
等价地，$u^+(s,a)=\log(\cand^+/\beh)$ 满足
\begin{equation}
\label{eq:optimal-logratio}
 u^+(s,a)=\eta f(s,a)-\log Z_f(s).
\end{equation}
\end{theorem}

\begin{proof}
先检查式 \eqref{eq:gibbs} 确实定义了概率分布。由于 $Z_f(s)$ 是有限正数，
\[
\sum_a\cand^+(a\mid s)
=\frac{1}{Z_f(s)}\sum_a\beh(a\mid s)e^{\eta f(s,a)}=1.
\]
并且 $\cand^+(a\mid s)>0$ 只可能发生在 $\beh(a\mid s)>0$ 的 action 上，所以 $\cand^+\ll\beh$。
对任意满足 $p\ll\beh$ 的候选分布，由定义逐项展开可得
\begin{align*}
\KL(p\Vert\cand^+)
&=\E_p\left[\log\frac{p}{\cand^+}\right]\\
&=\E_p\left[\log\frac{p}{\beh}
      -\eta f+\log Z_f(s)\right]\\
&=\KL(p\Vert\beh)-\frac1\eta\E_p[f]+\log Z_f(s).
\end{align*}
将等式两边乘以 $-1/\eta$ 并整理，得到目标恒等式
\begin{equation*}
\E_p[f]-\frac1\eta\KL(p\Vert\beh)
=\frac1\eta\log Z_f(s)-\frac1\eta\KL(p\Vert\cand^+).
\end{equation*}
KL 散度满足 $\KL(p\Vert\cand^+)\geq0$，且等号当且仅当 $p=\cand^+$ 几乎处处成立。因此任意候选分布的目标值都不超过 $\frac1\eta\log Z_f(s)$，而 $p=\cand^+$ 达到该上界，故它是唯一最优解。
\end{proof}

\begin{corollary}[优势形式]
若 $f=A^\beh$ 且 $\E_{\beh}[A^\beh(s,\cdot)\mid s]=0$，则
\[
\cand^+(a\mid s)=\beh(a\mid s)\exp(\eta A^\beh(s,a)) / Z_A(s).
\]
若 $A^\beh$ 是与序列 PMD 一致的 soft advantage，则 $Z_A(s)=1$；若它是硬 reward advantage，$Z_A(s)$ 一般不等于 $1$。
\end{corollary}

\section{即时 reward 与完整序列 reward}
\label{sec:immediate-vs-return}

若 $f(s,a)=r_t(s,a)$，式 \eqref{eq:token-subproblem} 是局部 token reward 的镜像更新。若完整目标是
\[
R(y)=\sum_{t=1}^{T}r_t(s_t,a_t),
\]
则与 sequence-level PMD 一致的 token score 必须是
\[
Q^\beh(s_t,a_t)=r_t(s_t,a_t)+\E_\beh\left[\sum_{k>t}r_k(s_k,a_k)\mid s_t,a_t\right],
\]
再取相应 soft advantage。直接把 $r_t$ 放进式 \eqref{eq:gibbs} 会得到局部目标，而不是完整序列目标。per-token reward 消除了终端奖励的稀疏性，却没有消除 future reward 对全局最优条件分布的作用。

\chapter{从同一个最优化问题推导 PMD-Part、PMD-Mean、BPO 与 MDPO}
\label{ch:four-methods}

\section{共同的 token-level mirror 子问题}
\label{sec:ch2-common}

固定 prefix $s$，行为策略记为 $\mu(a\mid s)$，候选学生策略记为 $\pi_\theta(a\mid s)$，并定义
\[
u_\theta(s,a)=\log\frac{\pi_\theta(a\mid s)}{\mu(a\mid s)},
\qquad
 g_\theta(s,a)=\nabla_\theta\log\pi_\theta(a\mid s).
\]
统一用 $\eta>0$ 表示 Natural Gradient 的学习率。给定 token 信号 $f(s,a)$，四种方法都从同一个镜像子问题开始：
\begin{equation}
\label{eq:ch2-mirror}
\max_{p\ll\mu(\cdot\mid s)}
\left\{
\E_{a\sim p}[f(s,a)]
-\frac1\eta\KL(p\Vert\mu(\cdot\mid s))
\right\}.
\end{equation}
定义
\[
Z_f(s)=\E_{a\sim\mu(\cdot\mid s)}[e^{\eta f(s,a)}].
\]
对任意候选分布 $p$，直接展开 KL 得到
\begin{align*}
\KL(p\Vert\pi^+)
&=\E_p\left[\log\frac{p}{\mu}-\eta f+\log Z_f(s)\right]\\
&=\KL(p\Vert\mu)-\eta\E_p[f]+\log Z_f(s),
\end{align*}
其中
\[
\pi^+(a\mid s)=\frac{\mu(a\mid s)e^{\eta f(s,a)}}{Z_f(s)}.
\]
整理后，镜像目标满足
\begin{equation}
\label{eq:ch2-kl-decomp}
\E_p[f]-\frac1\eta\KL(p\Vert\mu)
=\frac1\eta\log Z_f(s)-\frac1\eta\KL(p\Vert\pi^+).
\end{equation}
因此唯一最优策略是 $\pi^+$，其最优 log-ratio 为
\begin{equation}
\label{eq:ch2-optimal-logratio}
u^+(s,a)=\log\frac{\pi^+(a\mid s)}{\mu(a\mid s)}
=\eta f(s,a)-\log Z_f(s).
\end{equation}

\subsection{为什么优势函数的行为均值为零}

条件 $\E_\mu[f(s,a)\mid s]=0$ 不是任意 reward 的性质。若 $f=A^\mu$ 是行为策略的优势函数，则
\[
A^\mu(s,a)=Q^\mu(s,a)-V^\mu(s),
\qquad
V^\mu(s)=\E_{a\sim\mu}[Q^\mu(s,a)\mid s],
\]
从而
\begin{align*}
\E_\mu[A^\mu(s,a)\mid s]
&=\E_\mu[Q^\mu(s,a)\mid s]-V^\mu(s)\\
&=0.
\end{align*}
若 $f$ 是即时 reward、教师分数或一般 OPD 信号，则必须把中心化作为显式定义。令
\[
\bar f(s)=\E_\mu[f(s,a)\mid s],
\qquad
\widetilde f(s,a)=f(s,a)-\bar f(s).
\]
减去只依赖 prefix 的常数不改变 Gibbs 策略，因为
\[
\frac{\mu(a\mid s)e^{\eta\widetilde f(s,a)}}
{\E_\mu[e^{\eta\widetilde f(s,\cdot)}\mid s]}
=
\frac{\mu(a\mid s)e^{\eta f(s,a)}}
{\E_\mu[e^{\eta f(s,\cdot)}\mid s]}.
\]
后文在 BPO 和 PMD-Mean 中使用 $\widetilde f$；PMD-Part 直接保留精确的 $\log Z_f(s)$。

\section{四种算法的逐步推导}
\label{sec:ch2-four-methods}

\subsection{PMD-Part：回归完整的最优 log-ratio}

把式 \eqref{eq:ch2-optimal-logratio} 当作 supervised target，定义
\[
h(s,a)=\eta f(s,a)-\log Z_f(s),
\qquad
q_\theta(s,a)=h(s,a)-u_\theta(s,a).
\]
PMD-Part 损失和下降方向分别为
\begin{align}
\mathcal L_{\mathrm{Part}}
&=\frac12\E_{d_\mu,\mu}[q_\theta(s,a)^2],\label{eq:ch2-part-loss}\\
-\nabla_\theta\mathcal L_{\mathrm{Part}}
&=\E_{d_\mu,\mu}[q_\theta(s,a)g_\theta(s,a)].\label{eq:ch2-part-gradient}
\end{align}
给定 batch $\{(s_i,a_i)\}_{i=1}^{B}$，其中 $a_i\sim\mu(\cdot\mid s_i)$，实际代码使用
\begin{equation}
\label{eq:ch2-part-empirical}
\widehat{\mathcal L}_{\mathrm{Part}}=\frac{1}{2B}\sum_{i=1}^{B}\left(\log\pi_\theta(a_i\mid s_i)-\log\mu(a_i\mid s_i)-h_i\right)^2,\quad\widehat G_{\mathrm{Part}}=\frac1B\sum_i(h_i-u_i)g_i,
\end{equation}
其中 $h_i=\eta f_i-\log Z_f(s_i)$、$u_i=\log\pi_\theta(a_i\mid s_i)-\log\mu(a_i\mid s_i)$，并对 $h_i$ 停止梯度。若不能精确枚举 $Z_f$，可用同一 prefix 的 $K$ 个行为 action 估计 $\widehat Z_f(s)=K^{-1}\sum_{j=1}^{K}\exp(\eta f(s,a_j))$；此时 target 估计误差会进入回归偏差。
这里的 target 常数 $\log Z_f(s)$ 固定了 action-independent 方向。

\subsection{MDPO：直接优化带 KL 正则的镜像目标}
\label{sec:ch2-mdpo-generic}

MDPO 保留式 \eqref{eq:ch2-mirror} 的原始目标，不先构造 log-ratio target。对固定 prefix，定义
\begin{equation}
\label{eq:ch2-mdpo-generic-objective}
\mathcal J_{\mathrm{MDPO}}(\theta)
=\E_{a\sim\pi_\theta}[f(s,a)]
-\frac{1}{\eta}\KL\left(\pi_\theta(\cdot\mid s)\Vert\mu(\cdot\mid s)\right).
\end{equation}
令 $\rho_\theta(s,a)=\pi_\theta(a\mid s)/\mu(a\mid s)$。若可以枚举词表，KL 项直接计算为
\begin{equation}
\label{eq:ch2-mdpo-full-kl}
\KL(\pi_\theta\Vert\mu)=\sum_a\pi_\theta(a\mid s)\log\rho_\theta(s,a).
\end{equation}
若训练样本来自 $a_i\sim\mu(\cdot\mid s_i)$，则每个样本的可计算标量为
\begin{equation}
\label{eq:ch2-mdpo-sample-integrand}
\widehat j_i(\theta)=\rho_i f_i-\frac{1}{\eta}\rho_i\log\rho_i,\qquad \rho_i=\exp\left(\log\pi_\theta(a_i\mid s_i)-\log\mu(a_i\mid s_i)\right).
\end{equation}
对 $f_i$、$\mu$ 和行为样本停止梯度，batch ascent 方向为
\begin{equation}
\label{eq:ch2-mdpo-generic-gradient}
\widehat G_{\mathrm{MDPO}}=\frac1B\sum_{i=1}^{B}\rho_i\left(f_i-\frac1\eta\log\rho_i\right)g_i,\qquad g_i=\nabla_\theta\log\pi_\theta(a_i\mid s_i).
\end{equation}
这里 $\rho_i$ 是 action-level importance ratio；如果 prefix 也来自 off-policy replay，还需在样本外乘 $W_{<t_i}$。MDPO 因而是四种方法中唯一直接对镜像目标做 ascent 的 inner，其他三种先构造 regression residual。

\subsection{PMD-Mean：中心化 target 加归一化投影}

PMD-Mean 不显式回归 $\log Z_f$，而令
\[
g_{\mathrm{Mean}}(s,a)=\eta\widetilde f(s,a)
\]
并求解
\begin{equation}
\label{eq:ch2-mean-loss}
\min_u\frac12\E_{d_\mu,\mu}[(u-g_{\mathrm{Mean}})^2]
\quad\text{s.t.}\quad
\E_{a\sim\mu(\cdot\mid s)}[e^{u(s,a)}]=1.
\end{equation}
固定 prefix 后，写出拉格朗日函数
\[
\mathcal J(u,\lambda)
=\frac12\sum_a\mu_a(u_a-g_a)^2
+\lambda\left(\sum_a\mu_a e^{u_a}-1\right).
\]
因为 $\mu_a>0$，一阶条件为
\[
u_a-g_a+\lambda e^{u_a}=0.
\]
令 $v_a=g_a-u_a$，则
\[
v_a e^{v_a}=\lambda e^{g_a}.
\]
由 Lambert-$W$ 的定义 $W(z)e^{W(z)}=z$，得到
\begin{equation}
\label{eq:ch2-mean-lambert}
u_{\mathrm{Mean}}^+(s,a)
=g_{\mathrm{Mean}}(s,a)
-W\left(\lambda(s)e^{g_{\mathrm{Mean}}(s,a)}\right),
\end{equation}
其中 $\lambda(s)$ 由归一化约束唯一确定。PMD-Mean 是投影问题；除特殊情形外，式 \eqref{eq:ch2-mean-lambert} 不等于精确 Gibbs log-ratio。

\paragraph{PMD-Mean 的 batch 估计量。}
在每个 prefix 采 $K$ 个 action，记 $r_{ij}=\log(T(a_{ij}\mid s_i)/\mu(a_{ij}\mid s_i))$，则
\begin{equation}
\widehat{\bar r}_i=\frac1K\sum_{j=1}^{K}r_{ij},\qquad \widehat{\widetilde r}_{ij}=r_{ij}-\widehat{\bar r}_i.
\end{equation}
对第 $i$ 个 prefix 的第 $j$ 个样本，回归目标为 $\eta\widehat{\widetilde r}_{ij}$，经验损失为
\begin{equation}
\label{eq:ch2-mean-empirical}
\widehat{\mathcal L}_{\mathrm{Mean}}=\frac{1}{2BK}\sum_{i=1}^{B}\sum_{j=1}^{K}\left(u_{ij}-\eta\widehat{\widetilde r}_{ij}\right)^2,\qquad u_{ij}=\log\frac{\pi_\theta(a_{ij}\mid s_i)}{\mu(a_{ij}\mid s_i)}.
\end{equation}
若使用完整 vocabulary，则把样本均值替换为 $\sum_a\mu(a\mid s_i)r(s_i,a)$；若只使用单个 action，中心化项不可由该样本无偏估计，必须使用额外 action、教师全分布或 control variate。

\subsection{BPO：对同一最优性方程做中心化 residual 回归}

先定义 $b_\theta(s,a)=\eta f(s,a)-u_\theta(s,a)$。令
\[
\bar b_\theta(s)=\E_{a\sim\mu}[b_\theta(s,a)\mid s].
\]
BPO 的 residual 是
\begin{equation}
\label{eq:ch2-bpo-residual}
\delta_\theta(s,a)=b_\theta(s,a)-\bar b_\theta(s).
\end{equation}
若使用中心化信号 $\widetilde f$，则
\[
\delta_\theta
=\eta\widetilde f-u_\theta+\E_\mu[u_\theta\mid s],
\qquad
\E_\mu[\delta_\theta\mid s]=0.
\]
完整 reverse-KL 的导数给出
\[
\nabla_\theta\delta_\theta(s,a)
=-\left(g_\theta(s,a)-\bar g_\theta(s)\right),
\qquad
\bar g_\theta(s)=\E_\mu[g_\theta(s,a)\mid s].
\]
所以未线性化 BPO 损失
\begin{equation}
\label{eq:ch2-bpo-loss}
\mathcal L_{\mathrm{BPO}}
=\frac{1}{2\eta}\E_{d_\mu,\mu}[\delta_\theta(s,a)^2]
\end{equation}
的下降方向为
\begin{equation}
\label{eq:ch2-bpo-gradient}
-\eta\nabla_\theta\mathcal L_{\mathrm{BPO}}
=\E_{d_\mu,\mu}
[\delta_\theta(s,a)(g_\theta(s,a)-\bar g_\theta(s))].
\end{equation}
给定每个 prefix 的 $K$ 个行为 action，BPO 的经验估计量可以写成
\begin{equation}
\label{eq:ch2-bpo-empirical}
\widehat{\mathcal L}_{\mathrm{BPO}}
=\frac{1}{2\eta BK}\sum_{i=1}^{B}\sum_{j=1}^{K}\\widehat{\\delta}_{ij}^{\,2},
\qquad
\widehat G_{\mathrm{BPO}}
=\frac{1}{BK}\sum_{i=1}^{B}\sum_{j=1}^{K}
\widehat{\delta}_{ij}\left(g_{ij}-\widehat{\bar g}_i\right),
\end{equation}
其中
\[
\widehat{\bar g}_i=\frac1K\sum_{j=1}^{K}g_{ij},\qquad
\widehat{\delta}_{ij}=\eta f_{ij}-u_{ij}-\frac1K\sum_{\ell=1}^{K}(\eta f_{i\ell}-u_{i\ell}).
\]
同一 prefix 内的 action 集合用于估计行为测度下的中心化项；只有一个 action 时，$\widehat{\bar g}_i$ 与 residual 均值不能由该条样本本身可靠估计。

\subsection{PMD-Mean 与 BPO 都中心化，但投影对象不同}
\label{sec:ch2-mean-bpo-difference}

两种方法都可以写出一个中心化的信号 $\widetilde f$，但中心化发生在不同的优化问题中。PMD-Mean 先把精确最优 log-ratio

\[
u^+(s,a)=\eta f(s,a)-\log Z_f(s)
\]

替换为可观测的中心化 target $\eta\widetilde f(s,a)$，然后在满足 $\E_\mu[e^{u}\mid s]=1$ 的函数集合上做平方距离投影。它的残差是

\[
r_{\mathrm{Mean}}(s,a)=\eta\widetilde f(s,a)-u_\theta(s,a),
\]

并没有把当前策略的 $u_\theta$ 再次中心化。

BPO 使用的是同一个镜像最优性方程的 centered residual。展开式 \eqref{eq:ch2-bpo-residual} 可得

\[
\delta_\theta(s,a)=\eta f(s,a)-u_\theta(s,a)
-\E_\mu[\eta f(s,\cdot)-u_\theta(s,\cdot)\mid s].
\]

若 $f$ 已经满足 $\E_\mu[f\mid s]=0$，这变为

\[
\delta_\theta(s,a)=\eta f(s,a)-u_\theta(s,a)+\E_\mu[u_\theta(s,\cdot)\mid s].
\]

因此二者的 residual 差为

\[
\delta_\theta-r_{\mathrm{Mean}}=\E_\mu[u_\theta(s,\cdot)\mid s]
\]

（在 $f$ 已中心化时）。这个差是只依赖 prefix 的常数，策略在精确解处可以通过归一化吸收它；但是在参数空间进行有限步回归时，它会改变梯度，因为 PMD-Mean 使用 $g_\theta$，而未线性化 BPO 对完整 reverse-KL 求导后使用 $g_\theta-\bar g_\theta$。所以“都出现 $\widetilde f$”只说明它们共享同一个 action-independent 不可辨识方向，并不意味着两个损失或梯度相同。

\begin{proposition}[PMD-Mean 与 BPO 的等价条件]
\label{prop:ch2-mean-bpo-equivalence}
固定一个 prefix，并假设 $\E_\mu[f\mid s]=0$。若同时满足

\[
\E_\mu[u_\theta\mid s]=0,
\qquad
\bar g_\theta(s)=\E_\mu[g_\theta(s,a)\mid s]=0,
\]

则 PMD-Mean 的 residual 与 BPO residual 相同，且两者的下降方向相同。策略归一化只给出 $\E_{\pi_\theta}[g_\theta]=0$，而不是所需的 $\E_\mu[g_\theta]=0$；同样，$\E_\mu[u_\theta\mid s]=0$ 也不是归一化约束的结果。因此在行为测度 $\mu$ 下，二者通常只共享精确零点而不共享有限步梯度。
\end{proposition}

\begin{proof}
在 $\E_\mu[f\mid s]=0$ 下，前面的恒等式给出
\[
\delta_\theta-r_{\mathrm{Mean}}=\E_\mu[u_\theta\mid s].
\]
若该均值为零，则 residual 相同。PMD-Mean 的下降方向为 $\E_\mu[r_{\mathrm{Mean}}g_\theta]$；BPO 的下降方向由式 \eqref{eq:ch2-bpo-gradient} 给出。若再有 $\bar g_\theta=0$，则
\[
\E_\mu[\delta_\theta(g_\theta-\bar g_\theta)]
=\E_\mu[r_{\mathrm{Mean}}g_\theta],
\]
故两者相同。反之，只要任一 action-independent 项没有被这两个条件消除，有限参数更新就会产生梯度差异。
\end{proof}

\section{OPD 的 inner--outer mirror 流程：教师并非每步都被完全拟合}
\label{sec:ch2-opd-instantiation}

本节把 OPD 写成一个明确的 outer--inner 算法。固定教师策略 $T(a\mid s)$，令第 $k$ 次 outer 迭代开始时的学生策略为 $\pi_k(a\mid s)$。假设在教师有正概率的 action 上，学生也有正概率，即 $T(\cdot\mid s)\ll\pi_k(\cdot\mid s)$。OPD 的逐 token 信号取为当前学生到教师的 log-ratio
\begin{equation}
\label{eq:ch2-opd-current-reward}
f_k(s,a)=\log\frac{T(a\mid s)}{\pi_k(a\mid s)}.
\end{equation}
这里的 reward 依赖 outer 迭代编号 $k$；它不是一个固定的、一次性定义的 $r_T$。它衡量教师相对于当前学生在 prefix $s$ 下对 action $a$ 的局部偏好。

\subsection{一个 outer step 的精确 mirror 解}

给定 $\pi_k$ 和 $f_k$，第 $k$ 次 mirror 子问题为
\begin{equation}
\label{eq:ch2-opd-mirror-step}
\pi_{k+1}^{\star}
=\argmax_{p\ll\pi_k}
\left\{
\E_{a\sim p}[f_k(s,a)]
-\frac{1}{\eta}\KL\left(p\middle\Vert\pi_k(\cdot\mid s)\right)
\right\},
\qquad 0<\eta\leq 1.
\end{equation}
把 $f_k$ 代入 Gibbs 解，得到
\begin{align}
\pi_{k+1}^{\star}(a\mid s)
&=\frac{\pi_k(a\mid s)\exp\left(\eta\log\frac{T(a\mid s)}{\pi_k(a\mid s)}\right)}{Z_k(s)}\\
&=\frac{\pi_k(a\mid s)^{1-\eta}T(a\mid s)^{\eta}}{Z_k(s)},
\label{eq:ch2-opd-geometric-bridge}
\end{align}
其中
\begin{equation}
\label{eq:ch2-opd-bridge-normalizer}
Z_k(s)=\sum_{a}\pi_k(a\mid s)^{1-\eta}T(a\mid s)^{\eta}.
\end{equation}
因此，$\eta=1$ 时 $\pi_{k+1}^{\star}=T$，一次 mirror step 直接到教师；$0<\eta<1$ 时得到学生和教师在 log-probability 空间的几何插值。该中间分布不是额外启发式，而是式 \eqref{eq:ch2-opd-mirror-step} 的精确解。

\begin{theorem}[OPD mirror step 的几何中间分布]
\label{thm:ch2-opd-geometric-bridge}
在有限 action space 和 $T(\cdot\mid s)\ll\pi_k(\cdot\mid s)$ 下，式 \eqref{eq:ch2-opd-mirror-step} 的唯一解是式 \eqref{eq:ch2-opd-geometric-bridge}。特别地，$\eta=1$ 时一步到达教师；$0<\eta<1$ 时该更新保持在学生与教师之间，并满足 $\pi_{k+1}^{\star}\ll\pi_k$。
\end{theorem}

\begin{proof}
将 $f_k=\log(T/\pi_k)$ 代入镜像目标，并忽略与 $p$ 无关的项，有
\begin{align*}
\E_p[f_k]-\frac1\eta\KL(p\Vert\pi_k)
&=-\frac1\eta\E_p\left[\log p-(1-\eta)\log\pi_k-\eta\log T\right].
\end{align*}
令 $q_k(a\mid s)=\pi_k(a\mid s)^{1-\eta}T(a\mid s)^\eta$，并令 $Z_k=\sum_a q_k(a\mid s)$。则上式等于
\[
\frac1\eta\log Z_k-\frac1\eta\KL(p\Vert q_k/Z_k).
\]
KL 非负且仅在 $p=q_k/Z_k$ 时取零，因此得到唯一解。$\eta=1$ 时 $q_k=T$ 且 $Z_k=1$；当 $0<\eta<1$ 时，$q_k$ 对两种分布的 log 概率作凸组合，且由支持假设满足绝对连续性。
\end{proof}

\subsection{OPD token reward 的来源与序列层解释}
\label{sec:ch2-opd-reward-origin}

OPD 中的教师通常能在给定 prefix $s_t=(x,a_{<t})$ 时返回下一个 token 的条件分布 $T(a_t\mid s_t)$。学生在同一个 prefix 上给出 $\pi_k(a_t\mid s_t)$。两者的逐 token 对数似然差定义为
\begin{equation}
\label{eq:ch2-opd-token-logratio}
r_k(s_t,a_t)=\log T(a_t\mid s_t)-\log\pi_k(a_t\mid s_t).
\end{equation}
这个定义只使用教师和学生分布，不引入外部 reward。它也不是人为指定的常数，因为对一条完整序列 $y=(a_1,\ldots,a_H)$ 有
\begin{align}
\sum_{t=1}^{H}r_k(s_t,a_t)
&=\sum_{t=1}^{H}\log\frac{T(a_t\mid s_t)}{\pi_k(a_t\mid s_t)}\\
&=\log\frac{T(y\mid x)}{\pi_k(y\mid x)}.
\label{eq:ch2-opd-sequence-telescope}
\end{align}
因此 token reward 是序列级 teacher--student log-ratio 的一个无损分解。它把一个序列信号分摊到教师已经访问到的每个 prefix 上；它没有把序列目标替换成与未来无关的即时 reward。若优化的是完整序列目标，式 \eqref{eq:ch2-opd-sequence-telescope} 的后续 token 贡献仍然通过 return 或 soft $Q$ 进入 prefix 的训练权重。

\begin{proposition}[OPD reward 的镜像最优性]
\label{prop:ch2-opd-reward-optimality}
固定 prefix $s$ 和行为策略 $\pi_k$。令 $r_k$ 如式 \eqref{eq:ch2-opd-token-logratio}，则
\[
\pi_k(a\mid s)\exp\bigl(\eta r_k(s,a)\bigr)
=\pi_k(a\mid s)^{1-\eta}T(a\mid s)^\eta.
\]
所以以 $r_k$ 为 token 信号的 KL mirror 子问题，其最优分布必然是学生和教师的几何中间分布，而不是预先假设的教师分布。
\end{proposition}

\begin{proof}
将式 \eqref{eq:ch2-opd-token-logratio} 代入左侧：
\[
\pi_k(a\mid s)\exp\!\left(\eta\log\frac{T(a\mid s)}{\pi_k(a\mid s)}\right)
=\pi_k(a\mid s)^{1-\eta}T(a\mid s)^\eta.
\]
对 action 求和得到归一化常数 $Z_k(s)$，再应用定理 \ref{thm:gibbs} 即得结论。
\end{proof}

\subsection{同一个 OPD mirror step 的四种 inner 实例化}
\label{sec:ch2-opd-four-inner}

令
\[
u_k^\star(s,a)
=\log\frac{\pi_{k+1}^\star(a\mid s)}{\pi_k(a\mid s)}
=\eta r_k(s,a)-\log Z_k(s),
\]
其中
\[
r_k(s,a)=\log\frac{T(a\mid s)}{\pi_k(a\mid s)},
\qquad
Z_k(s)=\sum_{a'}\pi_k(a'\mid s)^{1-\eta}T(a'\mid s)^\eta.
\]
因此三个量都可以完全写成
\begin{equation}
\label{eq:ch2-opd-explicit-target}
u_k^\star(s,a)
=\eta\log\frac{T(a\mid s)}{\pi_k(a\mid s)}
-\log\sum_{a'}\pi_k(a'\mid s)^{1-\eta}T(a'\mid s)^\eta,
\qquad
u_\theta(s,a)=\log\frac{\pi_\theta(a\mid s)}{\pi_k(a\mid s)}.
\end{equation}
下面四种方法的差异只在于如何用有限样本和参数化模型逼近这个同一个 $u_k^\star$。

\paragraph{PMD-Part 的 OPD inner。}
若教师能够返回完整 token 分布，就可以直接计算
\[
h_k(s,a)
=\eta\log\frac{T(a\mid s)}{\pi_k(a\mid s)}
-\log\sum_{a'}\pi_k(a'\mid s)^{1-\eta}T(a'\mid s)^\eta.
\]
在 replay prefix 上最小化
\begin{equation}
\label{eq:ch2-opd-part-inner}
\mathcal L_{k,\mathrm{Part}}(\theta)
=\frac12\E_{s\sim d_b,\,a\sim\pi_k}
\left[
\log\frac{\pi_\theta(a\mid s)}{\pi_k(a\mid s)}
-\eta\log\frac{T(a\mid s)}{\pi_k(a\mid s)}
+\log\sum_{a'}\pi_k(a'\mid s)^{1-\eta}T(a'\mid s)^\eta
\right]^2.
\end{equation}
停止 target 梯度后，
\begin{equation}
\label{eq:ch2-opd-part-inner-gradient}
-\nabla_\theta\mathcal L_{k,\mathrm{Part}}
=\E_{d_b,\pi_k}
\left[h_k(s,a)-u_\theta(s,a)\right]g_\theta(s,a).
\end{equation}
对 batch $\{(s_i,a_i)\}_{i=1}^{B}$，其中 $s_i$ 来自 replay、$a_i\sim\pi_k(\cdot\mid s_i)$，对应的可执行估计量为
\begin{equation}
\label{eq:ch2-opd-part-empirical}
\widehat{\mathcal L}_{k,\mathrm{Part}}
=\frac{1}{2B}\sum_{i=1}^{B}(u_{\theta,i}-h_{k,i})^2,
\qquad
-\widehat{\nabla}\mathcal L_{k,\mathrm{Part}}
=\frac1B\sum_{i=1}^{B}(h_{k,i}-u_{\theta,i})g_{\theta,i}.
\end{equation}
如果 replay 的 prefix 测度是 $d_b$ 而目标 outer 测度是 $d_{\pi_k}$，把每一项替换为 $W^{k}_{<t_i}(u_{\theta,i}-h_{k,i})^2$，其中 $W^{k}_{<t_i}=d_{\pi_k}(s_i)/d_b(s_i)$ 的轨迹乘积估计见式 \eqref{eq:ch2-mdpo-prefix-weight}。
在 tabular、全 action 覆盖且使用真实 $Z_k$ 的情形，零损失等价于 $\pi_\theta=\pi_{k+1}^\star$。共享参数或 prefix 覆盖不足时，得到的是 replay 测度下的投影，不能把零训练误差解释成所有 prefix 上的教师拟合。

\paragraph{PMD-Mean 的 OPD inner。}
当 $Z_k(s)$ 不方便计算时，先构造行为测度下的中心化信号
\[
\widetilde r_k(s,a)
=\log\frac{T(a\mid s)}{\pi_k(a\mid s)}
-\sum_{a'}\pi_k(a'\mid s)
\log\frac{T(a'\mid s)}{\pi_k(a'\mid s)}.
\]
再令 $\widetilde h_k(s,a)=\eta\widetilde r_k(s,a)$，求解
\begin{equation}
\label{eq:ch2-opd-mean-inner}
\min_\theta\frac12\E_{d_b,\pi_k}
\left[
\log\frac{\pi_\theta(a\mid s)}{\pi_k(a\mid s)}
-\eta\log\frac{T(a\mid s)}{\pi_k(a\mid s)}
+\eta\sum_{a'}\pi_k(a'\mid s)
\log\frac{T(a'\mid s)}{\pi_k(a'\mid s)}
\right]^2
\quad\text{s.t.}\quad
\sum_a\pi_k(a\mid s)
\frac{\pi_\theta(a\mid s)}{\pi_k(a\mid s)}=1.
\end{equation}
在同一 prefix 采 $K$ 个 action 后，实际的 regression batch 为
\begin{equation}
\label{eq:ch2-opd-mean-empirical}
\widehat{\bar r}_{k,i}=\frac1K\sum_{j=1}^{K}\log\frac{T(a_{ij}\mid s_i)}{\pi_k(a_{ij}\mid s_i)},
\quad
\widehat{\mathcal L}_{k,\mathrm{Mean}}
=\frac{1}{2BK}\sum_{i,j}
\left[u_{\theta,ij}-\eta\left(r_{k,ij}-\widehat{\bar r}_{k,i}\right)\right]^2.
\end{equation}
其梯度估计为 $-\widehat{\nabla}\mathcal L_{k,\mathrm{Mean}}=(BK)^{-1}\sum_{i,j}\left[\eta(r_{k,ij}-\widehat{\bar r}_{k,i})-u_{\theta,ij}\right]g_{\theta,ij}$，并对 $\widehat{\bar r}_{k,i}$ 停止梯度。若目标状态测度为 $d_{\pi_k}$，每个 prefix loss 和梯度项再乘 $W^k_{<t_i}$。
约束是必要的：$u_\theta=\log(\pi_\theta/\pi_k)$ 必须满足
\[
\E_{\pi_k}[e^{u_\theta}\mid s]
=\sum_a\pi_\theta(a\mid s)=1.
\]
在函数空间中，式 \eqref{eq:ch2-opd-mean-inner} 是中心化 target 到这个非线性约束集合的投影；其 KKT 方程和 Lambert-$W$ 解已经在式 \eqref{eq:ch2-mean-lambert} 中给出。只有当中心化 target 本身满足该约束时，PMD-Mean 才直接等于精确 Gibbs log-ratio。

\paragraph{BPO 的 OPD inner。}
BPO 不显式估计 $\log Z_k$，而在每个 prefix 上把 residual 投影到行为测度的零均值子空间：
\begin{equation}
\delta_{k,\theta}(s,a)
\;=\eta\log\frac{T(a\mid s)}{\pi_k(a\mid s)}
-\log\frac{\pi_\theta(a\mid s)}{\pi_k(a\mid s)}
-\eta\sum_{a'}\pi_k(a'\mid s)
\log\frac{T(a'\mid s)}{\pi_k(a'\mid s)}
+\sum_{a'}\pi_k(a'\mid s)
\log\frac{\pi_\theta(a'\mid s)}{\pi_k(a'\mid s)}.
\label{eq:ch2-opd-bpo-inner-residual}
\end{equation}
\begin{equation}
\mathcal L_{k,\mathrm{BPO}}(\theta)
=\frac{1}{2\eta}\E_{d_b,\pi_k}\left[\delta_{k,\theta}(s,a)^2\right].
\label{eq:ch2-opd-bpo-inner-loss}
\end{equation}
对完整 reverse-KL 项求导得到
\begin{equation}
\label{eq:ch2-opd-bpo-inner-gradient}
-\eta\nabla_\theta\mathcal L_{k,\mathrm{BPO}}
=\E_{d_b,\pi_k}
\left[\delta_{k,\theta}(s,a)
\left(g_\theta(s,a)-\bar g_\theta(s)\right)\right],
\end{equation}
给定每个 prefix 的 $K$ 个 action，令
\[
\widehat{\bar g}_{\theta,i}=\frac1K\sum_{j=1}^{K}g_{\theta,ij},\qquad
\widehat{\delta}_{k,ij}=\eta r_{k,ij}-u_{\theta,ij}-\frac1K\sum_{\ell=1}^{K}(\eta r_{k,i\ell}-u_{\theta,i\ell}).
\]
则 BPO 的可执行 loss 和 ascent 估计量为
\begin{equation}
\label{eq:ch2-opd-bpo-empirical}
\widehat{\mathcal L}_{k,\mathrm{BPO}}=\frac{1}{2\eta BK}\sum_{i,j}\widehat{\delta}_{k,ij}^{,2},
\qquad
-\eta\widehat{\nabla}\mathcal L_{k,\mathrm{BPO}}
=\frac1{BK}\sum_{i,j}\widehat{\delta}_{k,ij}(g_{\theta,ij}-\widehat{\bar g}_{\theta,i}).
\end{equation}
若使用 replay prefix 测度近似 $d_{\pi_k}$，同样在每个 $i$ 的项前乘 $W^k_{<t_i}$；若 $K=1$，中心化 residual 和 centered score 都退化，不能从单个 action 无偏恢复 BPO 的行为条件中心化。
其中方括号表示标量 residual 与向量 score 的乘积，
\[
\delta_{k,\theta}(s,a)
\left(g_\theta(s,a)-\bar g_\theta(s)\right),
\qquad
\bar g_\theta(s)=\E_{\pi_k}[g_\theta(s,a)\mid s].
\]
若 $\delta_{k,\theta}=0$，则
\[
u_\theta(s,a)=\eta r_k(s,a)+c_k(s)
\]
对某个 action-independent 常数 $c_k(s)$ 成立。归一化约束进一步给出
\[
1=\E_{\pi_k}[e^{u_\theta}\mid s]
=e^{c_k(s)}Z_k(s),
\qquad
c_k(s)=-\log Z_k(s),
\]
所以 BPO 的精确零点也是 $u_k^\star$。这说明 BPO 与 PMD-Part 的最优策略相同，但其有限步梯度仍因中心化 score 而不同。

\input{sections/opd-mdpo-inner.tex}

\begin{theorem}[四种 OPD inner 的精确零点]
\label{thm:ch2-opd-inner-common-root}
在有限 action space、$\pi_k(a\mid s)>0$ 且函数类能够表示任意 log-ratio 的条件下，PMD-Part、PMD-Mean、未线性化 BPO 和 MDPO 的精确零点都对应
\[
\pi_\theta(\cdot\mid s)=\pi_{k+1}^{\star}(\cdot\mid s).
\]
它们的有限样本或有限参数更新不同，来自投影测度、target 常数的处理、score 是否中心化以及 MDPO 的 action ratio，而不是来自不同的 token-level mirror 最优解。
\end{theorem}

\begin{proof}
PMD-Part 的结论由 $u_\theta=h_k=u_k^\star$ 直接得到。BPO 的零点由上面的 residual 方程给出；归一化唯一确定 $c_k=-\log Z_k$，从而 $u_\theta=u_k^\star$。PMD-Mean 的可行集包含所有满足 $\E_{\pi_k}[e^u\mid s]=1$ 的 log-ratio；当中心化 target 可实现且投影残差为零时，它也得到同一 $u_k^\star$。对 MDPO，式 \eqref{eq:ch2-opd-mdpo-inner-full} 正是式 \eqref{eq:ch2-opd-mirror-step} 的参数化版本，利用 KL 分解可写成
\[
\mathcal J_{k,\mathrm{MDPO}}(\theta)
=\frac{1}{\eta}\log Z_k(s)
-\frac{1}{\eta}\KL(\pi_\theta\Vert\pi_{k+1}^{\star}),
\]
所以唯一最大点为 $\pi_\theta=\pi_{k+1}^{\star}$。若中心化 target 不可行或模型类受限，PMD-Mean 的解是约束集合上的投影，此时只能保证投影误差而不能保证精确 Gibbs 解。
\end{proof}

\subsection{MDPO 的 off-policy 权重阶梯}
\label{sec:ch2-mdpo-offpolicy-weights}

需要先区分 MDPO inner 的固定状态测度和完整 RL 目标的状态测度。第 $k$ 个 OPD inner 固定 anchor $q_k=\pi_k$，在每个 prefix 上优化
\begin{equation}
\label{eq:ch2-mdpo-fixed-state-objective}
\mathcal J_k(\theta)
=\sum_t\E_{s_t\sim d_{q_k}}
\left[
\E_{a\sim\pi_\theta(\cdot\mid s_t)}r_k(s_t,a)
-\frac1\eta\KL(\pi_\theta(\cdot\mid s_t)\Vert q_k(\cdot\mid s_t))
\right].
\end{equation}
这里的 $r_k(s,a)$ 是已经在当前 prefix--action 上定义好的 OPD token reward；它不是一条还要沿后缀重新采样的终端回报。设 replay 行为策略为 $b$，并令
\[
W^k_{<t}=\prod_{j<t}\frac{q_k(y_j\mid s_j)}{b(y_j\mid s_j)},
\qquad
\rho^{\theta/b}_t=\frac{\pi_\theta(y_t\mid s_t)}{b(y_t\mid s_t)}.
\]
则联合状态--动作测度满足
\begin{equation}
\label{eq:ch2-mdpo-prefix-weight}
\frac{d_{q_k}(s_t)\pi_\theta(a\mid s_t)}{d_b(s_t)b(a\mid s_t)}
=W^k_{<t}\rho^{\theta/b}_t.
\end{equation}
因此式 \eqref{eq:ch2-mdpo-fixed-state-objective} 的精确 off-policy 梯度估计量是
\begin{equation}
\label{eq:ch2-mdpo-offpolicy-gradient}
\widehat G_{k,\mathrm{MDPO}}
=\frac1B\sum_{i=1}^{B}
W^k_{<t_i}\rho^{\theta/b}_{i}
\left[
 r_{k,i}-\frac1\eta\log\frac{\pi_\theta(y_{t_i}\mid s_{t_i})}{q_k(y_{t_i}\mid s_{t_i})}
\right]g_{\theta,i}.
\end{equation}

\begin{proposition}[MDPO inner 的权重阶梯]
\label{prop:ch2-mdpo-weight-choice}
在式 \eqref{eq:ch2-mdpo-fixed-state-objective} 的固定 $d_{q_k}$ 目标下：
\begin{enumerate}[label=(\roman*)]
    \item 只用 $\rho_t^{\theta/b}$ 得到 $d_b$ 加权的镜像梯度，动作测度已修正但 prefix 访问测度仍有偏差；
    \item 使用 $W^k_{<t}\rho_t^{\theta/b}$ 恰好得到 $d_{q_k}$ 加权、$\pi_\theta$ 动作测度的镜像梯度；
    \item 再乘 $W_{>t}^{\theta/b}$ 会把局部 $r_k$ 的后缀测度改成候选策略 continuation，改变了目标，不再是式 \eqref{eq:ch2-mdpo-fixed-state-objective} 的估计量；
    \item 使用完整 $W_{1:T}^{\theta/b}$ 只在 $r_k$ 被替换为基于终端 $R(y)$ 的 continuation return、且目标本身要求对整条 $\pi_\theta$ 轨迹取期望时才是正确的 sequence IS。
\end{enumerate}
\end{proposition}

\begin{proof}
对任意只依赖 $(s_t,a_t)$ 的可积函数 $\varphi_t$，条件换测度给出
\begin{align*}
&\E_{y\sim b}\left[W^k_{<t}\rho_t^{\theta/b}\varphi_t(s_t,y_t)\right]\\
&=\sum_{s_t}d_b(s_t)W^k_{<t}(s_t)
  \sum_a b(a\mid s_t)\frac{\pi_\theta(a\mid s_t)}{b(a\mid s_t)}\varphi_t(s_t,a)\\
&=\sum_{s_t}d_{q_k}(s_t)\sum_a\pi_\theta(a\mid s_t)\varphi_t(s_t,a).
\end{align*}
令 $\varphi_t=(r_k-\eta^{-1}\log(\pi_\theta/q_k))g_t$ 即得式 \eqref{eq:ch2-mdpo-offpolicy-gradient}。如果只乘当前比率，第一行中的 $d_b(s_t)$ 不会变成 $d_{q_k}(s_t)$，所以残留的是 prefix occupancy bias。若再乘后缀比率，条件期望变为对 $\pi_\theta$ continuation 的积分；这只有在被积信号包含后缀随机变量、并且目标定义确实要求该 continuation 时才是合法换测度。
完整 $W_{1:T}^{\theta/b}$ 的合法性来自
\[
\E_b[W_{1:T}^{\theta/b}R(y)]=\E_{\pi_\theta}[R(y)],
\]
但它估计的是完整轨迹目标，而不是固定 $d_{q_k}$ 的局部 mirror inner。
\end{proof}

\paragraph{OPD 中的常用特例。}
若 replay 正好由 anchor $b=q_k=\pi_k$ 生成，则 $W^k_{<t}=1$，而
\[
\rho_t^{\theta/b}=\frac{\pi_\theta(y_t\mid s_t)}{\pi_k(y_t\mid s_t)}.
\]
此时 MDPO 只需当前 action ratio；这就是代码中常见的 MDPO inner 形式。若 replay 来自更旧策略或混合策略，才需要显式 prefix ratio $W^k_{<t}$。这与序列策略梯度的 $W_{\le t}$ 不矛盾：$W_{\le t}$ 的当前因子是把动作分布从 $b$ 换到目标策略，而 MDPO 的状态 anchor 仍是 $q_k$，所以当前因子是 $\pi_\theta/b$，前缀因子是 $q_k/b$。

\subsection{inner 求解与 outer 更新的区别}

式 \eqref{eq:ch2-opd-mirror-step} 是 outer 第 $k$ 步要解决的目标，不能把它和参数优化步数混为一谈。实际学生只能表示 $\pi_\theta$，因此固定行为策略 $\pi_k$ 后，inner 优化是在同一批 prefix 上最小化
\begin{equation}
\label{eq:ch2-opd-inner-regression}
\mathcal L_k(\theta)
=\frac12\E_{s\sim d_{\mathrm{replay}},\,a\sim\pi_k(\cdot\mid s)}
\left[
\log\frac{\pi_{k+1}^{\star}(a\mid s)}{\pi_k(a\mid s)}
-\log\frac{\pi_\theta(a\mid s)}{\pi_k(a\mid s)}
\right]^2.
\end{equation}
若 inner 被精确求解，得到 $\pi_\theta=\pi_{k+1}^{\star}$；若只做有限次 SGD，得到的是近似解 $\widehat\pi_{k+1}$，其误差会进入 outer 迭代的误差项。因而一个完整的 outer step 是：冻结 $\pi_k$，构造 $f_k$，用 replay 中丰富的 prefix 近似 $d_{\pi_k}$ 或其覆盖测度，进行若干 inner 更新，得到 $\widehat\pi_{k+1}$，然后令 $\pi_{k+1}\leftarrow\widehat\pi_{k+1}$ 并重新采样或继续积累数据。

\begin{proposition}[有限 inner 误差的 projective outer 递推]
\label{prop:ch2-opd-inner-error}
对每个 prefix 定义 log-ratio 振幅
\[
\Delta_k(s)=\max_a\log\frac{\pi_k(a\mid s)}{T(a\mid s)}-\min_a\log\frac{\pi_k(a\mid s)}{T(a\mid s)}.
\]
若精确 mirror 解被有限 inner 优化替换为 $\widehat\pi_{k+1}$，定义其 projective 误差为 $\zeta_k(s)$，使得
\[
\Delta(\widehat\pi_{k+1},T;s)\leq(1-\eta)\Delta_k(s)+\zeta_k(s).
\]
则
\begin{equation}
\label{eq:ch2-opd-outer-error}
\Delta_{k+1}(s)\leq(1-\eta)\Delta_k(s)+\zeta_k(s).
\end{equation}
因此若 $\limsup_k\zeta_k(s)\leq\zeta_\infty(s)$，则
\[
\limsup_{k\to\infty}\Delta_k(s)\leq\frac{\zeta_\infty(s)}{\eta}.
\]
\end{proposition}

\begin{proof}
精确 mirror 解满足式 \eqref{eq:ch2-opd-projective-contraction}。有限 inner 误差的定义直接给出三角型递推式 \eqref{eq:ch2-opd-outer-error}。递推展开后，初始项按 $(1-\eta)^k$ 衰减，误差项由几何级数控制：
\[
\sum_{j=0}^{k-1}(1-\eta)^j\zeta_{k-1-j}(s)
\leq \frac{\sup_{j\geq k_0}\zeta_j(s)}{\eta}+o(1).
\]
取上极限即得结论。
\end{proof}

\paragraph{关于 $L_1$ 距离。} 上述 projective 递推是无条件成立的；若进一步假设所有迭代满足 $m\leq\pi_k(a\mid s)/T(a\mid s)\leq M$，则归一化和有限 action space 将 projective 误差连续地转化为 $L_1$ 误差，从而得到相应的 $L_1$ 几何收敛界。没有这样的概率比上下界时，不能直接把 $1-\eta$ 写成任意 $L_1$ 距离的收缩系数。令
\[
\Delta_k(s)=\max_a\log\frac{\pi_k(a\mid s)}{T(a\mid s)}-\min_a\log\frac{\pi_k(a\mid s)}{T(a\mid s)},
\]
则精确满足
\begin{equation}
\label{eq:ch2-opd-projective-contraction}
\Delta_{k+1}(s)=(1-\eta)\Delta_k(s).
\end{equation}
由 $\E_T[\pi_k/T]=1$，$\Delta_k\to0$ 蕴含 $\pi_k\to T$。文中的 $L_1$ 递推应理解为在概率比有统一上下界时的推论；否则应优先使用式 \eqref{eq:ch2-opd-projective-contraction}。

\subsection{OPD 的 off-policy prefix 复用}

OPD 的优势在于教师可以在学生访问到的 prefix 上提供稠密 token 信号。设 replay 测度为 $d_b$，只要 $d_{\pi_k}\ll d_b$，outer mirror 目标可以写成
\begin{equation}
\label{eq:ch2-opd-prefix-is}
\E_{s\sim d_{\pi_k}}[\ell_k(s)]
=\E_{s\sim d_b}\left[W_{<t}(s)\ell_k(s)\right],
\qquad
W_{<t}(s)=\frac{d_{\pi_k}(s)}{d_b(s)}.
\end{equation}
PMD-Part、PMD-Mean 和 BPO 在 action 层都可用 $a\sim\pi_k(\cdot\mid s)$ 的样本完成回归，不需要额外 action-level IS；MDPO 则需要把行为 action 测度换成候选策略测度，因此保留 action ratio。对四种方法，off-policy 的 state 访问测度修正都由 prefix 权重 $W_{<t}$ 单独承担。若把 $d_b$ 当作固定投影测度而不做 $W_{<t}$，得到的是 replay 加权的近似 mirror step，而非严格的 $d_{\pi_k}$ 加权 outer 目标。这正是 OPD 可以利用状态丰富性、同时需要报告访问测度偏差的地方。

\section{共享零点、不同梯度的定理}
\label{sec:ch2-theorems}

\begin{proposition}[四种方法的共享零点]
\label{prop:ch2-shared-zero}
在有限 action space、$\mu(a\mid s)>0$ 且策略归一化成立时，PMD-Part、PMD-Mean、未线性化 BPO 与 MDPO 在精确 Gibbs 策略 $\pi^+$ 处达到零 residual。PMD-Mean 在其中心化 target 可实现并满足归一化约束时达到同一策略；一般情况下它给出约束集合上的欧氏投影。
\end{proposition}

\begin{proof}
PMD-Part 由 $u_\theta=h$ 直接得到 $u_\theta=u^+$。对 BPO，若 $\delta_\theta=0$，则 $u_\theta=\eta f-C(s)$，其中 $C(s)$ 与 action 无关。归一化给出
\[
1=\E_\mu[e^{u_\theta}\mid s]
=e^{-C(s)}\E_\mu[e^{\eta f}\mid s]
=e^{-C(s)}Z_f(s),
\]
所以 $C(s)=\log Z_f(s)$，即 $u_\theta=u^+$。PMD-Mean 在 target 可实现并满足归一化约束时也取到该策略；不可实现时，它取的是约束集合上的欧氏投影。
\end{proof}

\begin{theorem}[未线性化 BPO 与 PMD-Part 的梯度差异]
\label{thm:ch2-bpo-part-gap}
令 $q_\theta=h-u_\theta$，并令
\[
c_\theta(s)=\log Z_f(s)+\E_\mu[u_\theta(s,a)\mid s],
\qquad
\bar g_\theta(s)=\E_\mu[g_\theta(s,a)\mid s].
\]
则 $\delta_\theta=q_\theta+c_\theta$，且
\begin{equation}
\label{eq:ch2-bpo-part-gap}
-\eta\nabla\mathcal L_{\mathrm{BPO}}
=-\nabla\mathcal L_{\mathrm{Part}}
+\E_{s\sim d_\mu}[c_\theta(s)\bar g_\theta(s)].
\end{equation}
因此二者共享零点，但一般不具有相同的梯度场；只有 $c_\theta(s)\bar g_\theta(s)=0$ 对 $d_\mu$-几乎处处成立时才相同。
\end{theorem}

\begin{proof}
由 $q_\theta=\eta f-\log Z_f-u_\theta$，而 BPO 使用
$\delta_\theta=\eta f-u_\theta+\E_\mu[u_\theta\mid s]$，相减得到
$\delta_\theta=q_\theta+c_\theta$。另一方面，$\E_\mu[\delta_\theta\mid s]=0$，所以
\begin{align*}
-\eta\nabla\mathcal L_{\mathrm{BPO}}
&=\E[\delta_\theta(g_\theta-\bar g_\theta)]\\
&=\E[\delta_\theta g_\theta]\\
&=\E[(q_\theta+c_\theta)g_\theta]\\
&=\E[q_\theta g_\theta]+\E[c_\theta\bar g_\theta].
\end{align*}
第一项正是 $-\nabla\mathcal L_{\mathrm{Part}}$，得到式 \eqref{eq:ch2-bpo-part-gap}。
\end{proof}

\chapter{梯度的偏差与方差}
\label{ch:bias-variance}

\section{估计对象必须先固定}

固定 prefix $s$，令 $g(a)=\nabla_\theta\log\cand_\theta(a\mid s)$、$u(a)=\log(\cand_\theta(a\mid s)/\beh(a\mid s))$，并令 $\rho(a)=e^{u(a)}$。以下区分三个不同的随机量：

\begin{enumerate}[label=(\roman*)]
    \item 直接 KL 镜像目标的梯度 $\nabla F_s$；
    \item 行为测度下回归损失的梯度 $-\nabla\mathcal L_{\rm Part}$；
    \item BPO 的逐 prefix 残差损失梯度。
\end{enumerate}

三者在精确镜像解处共享零点，但只有第一个是 $F_s$ 的梯度。偏差必须相对于明确的目标定义；把行为测度下的回归梯度称为 ``无偏 off-policy policy gradient'' 会混淆两个目标。

\section{MDPO：无偏但承担 action-ratio 的二阶矩}

令 $c(a)=f(s,a)-\frac1\eta u(a)$。MDPO 的单样本估计量为
\begin{equation}
\widehat G_{\rm MDPO}=\rho(a)c(a)g(a),\qquad a\sim\beh(\cdot\mid s),
\end{equation}
并且在 $\cand\ll\beh$、$\E_\beh[\rho\lVert cg\rVert]<\infty$ 时
\begin{equation}
\E_\beh[\widehat G_{\rm MDPO}]=\E_\cand[cg]=\nabla F_s.
\end{equation}
因此 action-level IS 本身不产生偏差；若 prefix 来自 $d_\beh$ 而目标要求 $d_\cand$，还必须乘 $W_{<t}=d_\cand(s_t)/d_\beh(s_t)$。这两个密度比修正的是不同测度。

\begin{theorem}[MDPO 梯度的精确偏差--方差分解]
\label{thm:mdpo-exact-mse}
设 $G=\E_\beh[\rho cg]$，批量大小为 $n$，并假设 $\E_\beh[\rho^2c^2\lVert g\rVert^2]<\infty$。则样本均值 $\widehat G_n$ 满足
\begin{align}
\operatorname{Bias}(\widehat G_n)&=0,\\
\E\lVert\widehat G_n-G\rVert^2
&=\frac1n\left(\E_\beh[\rho^2c^2\lVert g\rVert^2]-\lVert G\rVert^2\right).
\end{align}
若 $\lVert g\rVert\leq B_g$、$|c|\leq B_c$，则
\begin{equation}
\operatorname{tr}\operatorname{Cov}(\widehat G_n)
\leq\frac{B_g^2B_c^2}{n}(1+\chi^2(\cand\Vert\beh)).
\end{equation}
\end{theorem}

\begin{proof}
令
\[
X_i=\rho(a_i)c(a_i)g(a_i),
\qquad a_i\stackrel{\mathrm{i.i.d.}}{\sim}\beh,
\qquad \widehat G_n=\frac1n\sum_{i=1}^nX_i.
\]
由 support 条件，$\rho=\cand/\beh$ 在 $\beh$-几乎处处有定义。换测度恒等式给出
\begin{align*}
\E_\beh[X_i]
&=\sum_a\beh(a)\frac{\cand(a)}{\beh(a)}c(a)g(a)\\
&=\sum_a\cand(a)c(a)g(a)=\E_\cand[cg]=G.
\end{align*}
因此
$\E[\widehat G_n]=n^{-1}\sum_i\E[X_i]=G$，样本均值的偏差严格为零。
再令 $Y_i=X_i-G$。独立性和 $\E[Y_i]=0$ 消掉交叉项，故
\begin{align*}
\E\lVert\widehat G_n-G\rVert^2
&=\E\left\lVert\frac1n\sum_iY_i\right\rVert^2\\
&=\frac{1}{n^2}\sum_i\E\lVert Y_i\rVert^2
=\frac1n\left(\E\lVert X_1\rVert^2-\lVert G\rVert^2\right).
\end{align*}
这就是向量均方误差公式。最后，若 $\lVert g\rVert\leq B_g$ 且 $|c|\leq B_c$，则
\[
\E_\beh\lVert X_1\rVert^2
\leq B_g^2B_c^2\E_\beh[\rho^2]
=B_g^2B_c^2\bigl(1+\chi^2(\cand\Vert\beh)\bigr),
\]
其中最后一个等号来自
$\chi^2(\cand\Vert\beh)=\E_\beh[(\rho-1)^2]=\E_\beh[\rho^2]-1$。去掉非负项 $\lVert G\rVert^2$ 即得到协方差迹的上界。
\end{proof}

在自回归序列中，$W_{\leq t}=\prod_{k\leq t}\rho_k$ 的二阶矩可以逐步展开。给定 prefix $y_{<t}$，有
\[
\E_\beh[\rho_t^2\mid y_{<t}]
=\sum_a\beh(a\mid y_{<t})
\left(\frac{\cand(a\mid y_{<t})}{\beh(a\mid y_{<t})}\right)^2
=1+\chi^2\bigl(\cand(\cdot\mid y_{<t})\Vert\beh(\cdot\mid y_{<t})\bigr).
\]
因为 $W_{\leq t}=W_{<t}\rho_t$，塔式法则给出
\begin{align*}
\E_\beh[W_{\leq t}^2]
&=\E_\beh\left[W_{<t}^2\E_\beh[\rho_t^2\mid y_{<t}]\right]\\
&=\E_\beh\left[W_{<t}^2\left(1+\chi^2(s_t)\right)\right].
\end{align*}
若所有可达 prefix 都满足 $c_{\rm lo}\leq\chi^2(s)\leq c_{\rm hi}$，则递推归纳得到
\begin{equation}
(1+c_{\rm lo})^t\leq\E_\beh[W_{\leq t}^2]\leq(1+c_{\rm hi})^t.
\end{equation}
这给出了 prefix IS 方差随修正长度增长的明确阶数。

再考虑完整权重 $W=W_{\leq t}W_{>t}$。在条件 $y_{\leq t}$ 下，后缀权重的均值为一：$\E[W_{>t}\mid y_{\leq t}]=1$，因此
$\E[W\mid y_{\leq t}]=W_{\leq t}$。全方差公式于是给出
\begin{equation}
\label{eq:prefix-total-variance}
\begin{aligned}
\operatorname{Var}(W)
&=\operatorname{Var}(\E[W\mid y_{\leq t}])
 +\E[\operatorname{Var}(W\mid y_{\leq t})]\\
&=\operatorname{Var}(W_{\leq t})
 +\E[W_{\leq t}^2\operatorname{Var}(W_{>t}\mid y_{\leq t})].
\end{aligned}
\end{equation}
第二项非负；只要 prefix 之后存在策略比率不恒等于一的可达状态，它就严格为正。这说明后缀权重只能增加方差，不能抵消 prefix 权重已经产生的方差。

\section{PMD-Part 的投影偏差与 BPO 的中心化方差}

设策略类受限为 $\Pi$。PMD-Part 的 population 解为
\begin{equation}
\theta_{\rm Part}^*=\argmin_{\theta\in\Pi}\E_{d_\beh,\beh}[(u_\theta-h)^2],
\end{equation}
而 BPO 的逐 prefix 解为
\begin{equation}
\theta_{\rm BPO}^*=\argmin_{\theta\in\Pi}\E_{d_\beh}\operatorname{Var}_{a\sim\beh}[u_\theta(a)-\eta f(a)].
\end{equation}
二者在可实现时拥有同一零点；不可实现时，BPO 不惩罚 action-independent 的误差，因此它允许更大的常数方向自由度，而 PMD-Part 通过精确的 $\log Z_f$ 固定该常数。共享参数网络下，$d_\beh$ 同时决定状态间投影误差如何分配。

\begin{proposition}[prefix IS 的偏差--方差阶梯]
\label{prop:prefix-bias-variance}
对终端或未来回报 $R$，行为样本下的 token-level 不加前缀权重估计量，其期望与目标策略梯度的差可分解为
\begin{equation}
\sum_s(d_\cand-d_\beh)(s)\sum_a\nabla\cand(a\mid s)A^\beh(s,a)
+\sum_s d_\cand(s)\sum_a\nabla\cand(a\mid s)(A^\cand-A^\beh)(s,a).
\end{equation}
乘 $W_{<t}$ 后第一项恰好消失；再乘后缀 $W_{>t}$ 才能修正第二项，但其额外方差正是式 \eqref{eq:prefix-total-variance} 的第二项。
\end{proposition}

\begin{proof}
第一项来自把 $d_\cand$ 替换为 $d_\beh$，第二项来自把 $A^\cand$ 替换为可估计的 $A^\beh$。前缀密度比将行为 prefix 测度换成 $d_\cand$，所以只消掉第一项；后缀比率不改变已经使用的优势估计，却引入条件二阶矩，故只能修正价值滞后而不能降低方差。
\end{proof}

\begin{remark}[可加残差为何能绕开乘积]
若信号写成逐 token 鞅差之和 $D=\sum_tD_t$，且 $\E[D_t\mid s_t]=0$，则交叉项消失并有
\begin{equation}
\E[D^2]=\sum_t\E[D_t^2].
\end{equation}
这正是离策略修正文档中 ``可加泛函的换测度界'' 的核心：回归型 PMD/BPO 利用逐点最优性或中心化残差，把序列乘积权重替换成逐步二阶量；代价是目标从真策略梯度改成行为测度下的投影问题。

\end{remark}

\begin{proposition}[可加 BPO 残差的损失估计方差]
\label{prop:bpo-loss-variance}
设轨迹残差可写成 $\delta=\sum_{t=1}^{T}e_t$，其中 $e_t$ 是关于 prefix 过滤的鞅差，且 $|e_t|\leq B_e$。则
\begin{equation}
\E[\delta^2]=\sum_{t=1}^{T}\E[e_t^2],
\qquad
\E[\delta^4]\leq T^2B_e^2\sum_{t=1}^{T}\E[e_t^2].
\end{equation}
对 $n$ 条独立轨迹的平方残差损失估计量
$\widehat{\mathcal L}=n^{-1}\sum_i\delta_i^2/(2\eta)$，有
\begin{equation}
\operatorname{Var}(\widehat{\mathcal L})
\leq \frac{T^2B_e^2}{4\eta^2n}
\sum_{t=1}^{T}\E[e_t^2].
\end{equation}
因此中心化残差的统计方差按 token 二阶量相加，而不是按 sequence ratio 的二阶矩相乘。
\end{proposition}

\begin{proof}
鞅差正交给出第一式。由 Cauchy--Schwarz，
$\delta^2=(\sum_te_t)^2\leq T\sum_te_t^2\leq T^2B_e^2$，故
$\delta^4\leq T^2B_e^2\delta^2$。独立样本均值的方差为单样本方差除以 $n$，再用
$\operatorname{Var}(\delta^2)\leq\E[\delta^4]$ 即得结论。
\end{proof}

\section{BPO 论文的训练预算}

\section{经典 MDPO 的训练预算与 IS}
\label{sec:mdpo-implementation}

经典 MDPO on-policy 代码默认 `sgd\_steps=5`，MuJoCo README 的推荐示例使用 10。一个 rollout batch 固定后，对同一批数据重复执行该数量的 policy optimizer updates，再刷新 old/new anchor；没有根据 mirror residual 自适应停止。

off-policy 代码的 README 暴露 `sgd\_steps=1000`，但当前训练循环实际写为 `for grad\_step in range(1)`，原本的 `int(self.gradient\_steps)` 被注释。因此每个 train event 实际只执行一次 actor/critic update；`gradient\_steps` 主要影响 target/anchor policy 同步周期。实现使用 replay 状态测度和 action-level policy gradient，不能解释为完整的 prefix-IS 求解。

\section{比较时应报告的指标}
\label{sec:reporting}

为了比较四种方法，实验至少应固定同一 rollout batch，并记录：每个 inner update 的 mirror loss、log-ratio residual norm、$\KL(\cand_\theta\Vert\beh)$、梯度范数、ratio 的均值与二阶矩、中心化 residual 的条件方差，以及 prefix coverage。这样才能区分“做了更多 optimizer steps”和“真正接近 token-level mirror optimum”。

\chapter{结论与 MirrorOPD 的边界}
\label{ch:conclusion}

从同一个 token-level Gibbs 最优解出发，PMD-Part 是精确 log-ratio 回归，PMD-Mean 是中心化回归后的 Lambert-$W$ 投影，MDPO 是带 action IS 的直接 KL 正则化 policy gradient。BPO 的完整 reverse-KL 平方 residual 在未线性化时产生中心化 score 梯度；它与 PMD-Part 共享可实现时的最优点，但一般相差 $c_s\bar g$，因此不能把二者的梯度直接视为相同。线性化、二元 KL 化简和工程截断属于后续近似，本文件没有把它们作为理论等价的前提。

本文件暂不推导 OPD 的完整 off-policy correction 或 KL 系数调度。后续 MirrorOPD 需要先选择局部即时 reward 目标还是完整加性回报目标，再决定是否引入 prefix IS、是否采用 Part/Mean 投影，以及如何用 Natural Gradient 控制学生策略与教师策略的距离。
