\chapter*{摘要}
\addcontentsline{toc}{chapter}{摘要}
本文从固定 prefix 上的 KL 镜像子问题出发，统一写出 PMD-Part、PMD-Mean、BPO 与 MDPO 的优化目标、精确最优解和参数梯度。主线结论是：PMD-Part 的闭式 target 是 Gibbs 分布；PMD-Mean 是带单纯形约束的 Lambert-$W$ 投影；MDPO 直接优化 KL 正则化目标，梯度中保留 action importance ratio；BPO 从平方 residual 回归出发；本稿保留完整 reverse-KL 导数并分析其中心化 score 梯度，不把后续线性化、二元 KL 或工程权重作为理论前提。四者可以共享同一个 token-level 最优策略，但离开最优点后的梯度并不相同。重要性采样降低测度偏差却引入 ratio 方差，回归型方法则承担访问测度下的投影偏差。最后核对 OpenKimi、经典 MDPO 和 BPO 论文源码，区分固定 inner budget 与真正的子问题收敛。

\chapter{统一的 token-level 镜像子问题}
\label{ch:unified-problem}

\section{Prefix、行为策略与局部 reward}
\label{sec:notation}

固定一个 prefix $s=(x,a_{<t})$，词表动作为 $a$，行为策略为 $\beh(a\mid s)$，待更新学生策略为 $\cand_\theta(a\mid s)$。以下所有结论都在
\[
\cand_\theta(\cdot\mid s)\ll\beh(\cdot\mid s)
\]
的 support 条件下成立。记局部监督信号为 $f(s,a)$；它可以是即时 per-token reward、优势 $A^\beh(s,a)$，也可以是包含未来 reward 的 soft $Q$。

设 $d_\beh(s)$ 是行为轨迹访问 prefix 的测度。若一个条件 target 在每个 prefix 上可以独立实现，则 $d_\beh(s)>0$ 只决定该 prefix 在总体训练损失中的权重；如果模型共享参数、数据有限或存在未覆盖 prefix，它会改变投影点和估计方差。

\section{KL 镜像最优解}
\label{sec:token-optimum}

给定 Natural Gradient 学习率 $\eta>0$，token-level PMD 子问题为
\begin{equation}
\label{eq:token-subproblem}
 \cand^+(\cdot\mid s)=\argmax_{p\ll\beh(\cdot\mid s)}
 \left\{\E_{a\sim p}[f(s,a)]-\frac{1}{\eta}\KL(p\Vert\beh(\cdot\mid s))\right\}.
\end{equation}
令
\[
 Z_f(s)=\E_{a\sim\beh(\cdot\mid s)}\exp\left(\eta f(s,a)\right).
\]

\begin{theorem}[Token-level Gibbs 解]
\label{thm:gibbs}
若 $0<Z_f(s)<\infty$，式 \eqref{eq:token-subproblem} 的唯一解为
\begin{equation}
\label{eq:gibbs}
 \cand^+(a\mid s)=\frac{\beh(a\mid s)\exp(\eta f(s,a))}{Z_f(s)}.
\end{equation}
等价地，$u^+(s,a)=\log(\cand^+/\beh)$ 满足
\begin{equation}
\label{eq:optimal-logratio}
 u^+(s,a)=\eta f(s,a)-\log Z_f(s).
\end{equation}
\end{theorem}

\begin{proof}
先检查式 \eqref{eq:gibbs} 确实定义了概率分布。由于 $Z_f(s)$ 是有限正数，
\[
\sum_a\cand^+(a\mid s)
=\frac{1}{Z_f(s)}\sum_a\beh(a\mid s)e^{\eta f(s,a)}=1.
\]
并且 $\cand^+(a\mid s)>0$ 只可能发生在 $\beh(a\mid s)>0$ 的 action 上，所以 $\cand^+\ll\beh$。
对任意满足 $p\ll\beh$ 的候选分布，由定义逐项展开可得
\begin{align*}
\KL(p\Vert\cand^+)
&=\E_p\left[\log\frac{p}{\cand^+}\right]\\
&=\E_p\left[\log\frac{p}{\beh}
      -\eta f+\log Z_f(s)\right]\\
&=\KL(p\Vert\beh)-\frac1\eta\E_p[f]+\log Z_f(s).
\end{align*}
将等式两边乘以 $-1/\eta$ 并整理，得到目标恒等式
\begin{equation*}
\E_p[f]-\frac1\eta\KL(p\Vert\beh)
=\frac1\eta\log Z_f(s)-\frac1\eta\KL(p\Vert\cand^+).
\end{equation*}
KL 散度满足 $\KL(p\Vert\cand^+)\geq0$，且等号当且仅当 $p=\cand^+$ 几乎处处成立。因此任意候选分布的目标值都不超过 $\frac1\eta\log Z_f(s)$，而 $p=\cand^+$ 达到该上界，故它是唯一最优解。
\end{proof}

\begin{corollary}[优势形式]
若 $f=A^\beh$ 且 $\E_{\beh}[A^\beh(s,\cdot)\mid s]=0$，则
\[
\cand^+(a\mid s)=\beh(a\mid s)\exp(\eta A^\beh(s,a)) / Z_A(s).
\]
若 $A^\beh$ 是与序列 PMD 一致的 soft advantage，则 $Z_A(s)=1$；若它是硬 reward advantage，$Z_A(s)$ 一般不等于 $1$。
\end{corollary}

\section{即时 reward 与完整序列 reward}
\label{sec:immediate-vs-return}

若 $f(s,a)=r_t(s,a)$，式 \eqref{eq:token-subproblem} 是局部 token reward 的镜像更新。若完整目标是
\[
R(y)=\sum_{t=1}^{T}r_t(s_t,a_t),
\]
则与 sequence-level PMD 一致的 token score 必须是
\[
Q^\beh(s_t,a_t)=r_t(s_t,a_t)+\E_\beh\left[\sum_{k>t}r_k(s_k,a_k)\mid s_t,a_t\right],
\]
再取相应 soft advantage。直接把 $r_t$ 放进式 \eqref{eq:gibbs} 会得到局部目标，而不是完整序列目标。per-token reward 消除了终端奖励的稀疏性，却没有消除 future reward 对全局最优条件分布的作用。

\chapter{统一的 token-level Mirror Descent 与 OPD 实例化}
\label{ch:four-methods}

本章只使用一套符号。固定第 $k$ 个 outer step 的 anchor policy $\pi_k$，候选 student policy 记为 $\pi_\theta$，teacher policy 记为 $T$。自然梯度步长统一记为 $\eta>0$。prefix 记为 $s$，下一个 token 记为 $a$，所有条件分布均省略 $s$ 时会在公式中明确补回。四种 inner 的区别只来自求解同一个 mirror 子问题的方式；它们不对应四个不同的 token-level 最优策略。

\section{共同的 mirror 子问题与精确解}
\label{sec:ch2-common}

给定固定 prefix $s$ 和 token 信号 $f(s,a)$，定义
\begin{equation}
\label{eq:ch2-mirror}
\mathcal M_s(p;f,\pi_k)
=\mathbb E_{a\sim p(\cdot\mid s)}[f(s,a)]
-\frac1\eta\operatorname{KL}\!\left(p(\cdot\mid s)\middle\Vert\pi_k(\cdot\mid s)\right).
\end{equation}
镜像子问题是在所有满足 $p\ll\pi_k$ 的概率分布上最大化 $\mathcal M_s$。定义
\begin{equation}
\label{eq:ch2-mirror-normalizer}
Z_f(s)=\sum_{v\in\mathcal V}\pi_k(v\mid s)\exp\!\bigl(\eta f(s,v)\bigr).
\end{equation}

\begin{theorem}[KL mirror 子问题的 Gibbs 解]
\label{thm:ch2-gibbs}
在有限词表、$Z_f(s)<\infty$ 且 $\pi_k(v\mid s)>0$ 的条件下，式 \eqref{eq:ch2-mirror} 的唯一解是
\begin{equation}
\label{eq:ch2-optimal-policy}
\pi_k^+(v\mid s)=\frac{\pi_k(v\mid s)e^{\eta f(s,v)}}{Z_f(s)},
\qquad
\log\frac{\pi_k^+(v\mid s)}{\pi_k(v\mid s)}
=\eta f(s,v)-\log Z_f(s).
\end{equation}
对任意候选 $p$，还有精确分解
\begin{equation}
\label{eq:ch2-kl-decomp}
\mathcal M_s(p;f,\pi_k)
=\frac1\eta\log Z_f(s)
-\frac1\eta\operatorname{KL}\!\left(p\middle\Vert\pi_k^+(\cdot\mid s)\right).
\end{equation}
\end{theorem}

\begin{proof}
由式 \eqref{eq:ch2-optimal-policy}，有
\[
\log\frac{p(v\mid s)}{\pi_k^+(v\mid s)}
=\log\frac{p(v\mid s)}{\pi_k(v\mid s)}-\eta f(s,v)+\log Z_f(s).
\]
对 $v$ 按 $p$ 求期望，得到
\[
\operatorname{KL}(p\Vert\pi_k^+)
=\operatorname{KL}(p\Vert\pi_k)-\eta\mathbb E_p[f]+\log Z_f(s).
\]
移项即得式 \eqref{eq:ch2-kl-decomp}。KL 非负且仅在 $p=\pi_k^+$ 时取零，因此最优解唯一。
\end{proof}

\begin{definition}[中心化 token 信号]
\label{def:ch2-centered-signal}
对任意 $f$ 定义 anchor 条件均值和中心化信号
\[
\bar f_k(s)=\mathbb E_{a\sim\pi_k}[f(s,a)\mid s],
\qquad
\widetilde f_k(s,a)=f(s,a)-\bar f_k(s).
\]
若 $f=A^{\pi_k}$ 是 anchor 的优势函数，则 $\bar f_k(s)=0$ 是优势函数定义推出的恒等式；对一般 teacher score 或 OPD log-ratio，$\bar f_k(s)=0$ 不是自动成立的假设。
\end{definition}

\begin{proposition}[中心化不改变 mirror 最优策略]
\label{prop:ch2-centering}
用 $\widetilde f_k$ 替代 $f$ 不改变式 \eqref{eq:ch2-mirror} 的最优分布。
\end{proposition}

\begin{proof}
因为 $\bar f_k(s)$ 与 action 无关，
\[
\pi_k(v\mid s)e^{\eta\widetilde f_k(s,v)}
=e^{-\eta\bar f_k(s)}\pi_k(v\mid s)e^{\eta f(s,v)}.
\]
分子和归一化常数同时乘以同一个因子 $e^{-\eta\bar f_k(s)}$，归一化后的分布不变。
\end{proof}

\section{四种 inner 的统一推导}
\label{sec:ch2-four-methods}

下面固定一个 prefix，令
\[
\ell^T(a)=\log T(a\mid s),\qquad
\ell^k(a)=\log\pi_k(a\mid s),\qquad
\ell^\theta(a)=\log\pi_\theta(a\mid s).
\]
训练时只对 $\ell^\theta$ 反向传播；teacher、anchor、目标常数和所有估计的 KL 都停止梯度。

\subsection{PMD-Part：回归完整 Gibbs log-ratio}
\label{sec:ch2-pmd-part}

定义显式 target
\[
H_k(s,a)=\eta f(s,a)-\log Z_f(s),
\qquad
U_\theta(s,a)=\log\pi_\theta(a\mid s)-\log\pi_k(a\mid s).
\]
PMD-Part 在 anchor 测度下最小化
\begin{equation}
\label{eq:ch2-part-loss}
\mathcal L_{\mathrm{Part}}(\theta)
=\frac12\mathbb E_{s\sim d_k,\,a\sim\pi_k}
\left[U_\theta(s,a)-H_k(s,a)\right]^2.
\end{equation}
对 target 停止梯度，链式法则给出
\begin{equation}
\label{eq:ch2-part-gradient}
-\nabla_\theta\mathcal L_{\mathrm{Part}}
=\mathbb E_{d_k,\pi_k}
\left[H_k(s,a)-U_\theta(s,a)\right]
\nabla_\theta\log\pi_\theta(a\mid s).
\end{equation}
这一步是普通的 log-ratio regression；它没有把 KL 视为常数，也没有使用 action-level importance ratio。

\subsection{PMD-Mean：中心化 target 与归一化约束}
\label{sec:ch2-pmd-mean}

由定义可得
\[
\mathbb E_{a\sim\pi_k}[f(s,a)]=\bar f_k(s),
\qquad
\mathbb E_{a\sim\pi_k}[\widetilde f_k(s,a)]=0.
\]
PMD-Mean 用 $\eta\widetilde f_k$ 作为回归 target，并要求 student log-ratio 满足归一化约束
\begin{equation}
\label{eq:ch2-mean-loss}
\min_{U}\ \frac12\mathbb E_{\pi_k}\left[(U-\eta\widetilde f_k)^2\mid s\right]
\quad\text{s.t.}\quad
\mathbb E_{\pi_k}[e^{U}\mid s]=1.
\end{equation}
约束正是 $\sum_a\pi_\theta(a\mid s)=1$ 在 log-ratio 坐标中的表达。若把词表上的函数值写为 $U_a$ 和 $G_a=\eta\widetilde f_k(s,a)$，拉格朗日函数为
\[
\mathscr L(U,\lambda)
=\frac12\sum_a\pi_k(a\mid s)(U_a-G_a)^2
+\lambda\left(\sum_a\pi_k(a\mid s)e^{U_a}-1\right).
\]
一阶条件为
\[
U_a-G_a+\lambda e^{U_a}=0.
\]
令 $V_a=G_a-U_a$，则 $V_a e^{V_a}=\lambda e^{G_a}$，所以
\begin{equation}
\label{eq:ch2-mean-lambert}
U_a=G_a-W\!\left(\lambda e^{G_a}\right),
\end{equation}
其中 $\lambda$ 由归一化约束唯一确定。PMD-Mean 因而是中心化 target 在指数归一化集合上的投影；只有当中心化 target 自身满足该约束时，投影才等于精确 Gibbs log-ratio。

\subsection{BPO：中心化最优性 residual}
\label{sec:ch2-bpo}

BPO 不显式回归 $\log Z_f$，而将同一个最优性方程的 action-independent 部分消去。定义
\begin{equation}
\label{eq:ch2-bpo-residual}
D_\theta(s,a)
=\eta f(s,a)-U_\theta(s,a)
-\mathbb E_{v\sim\pi_k}[\eta f(s,v)-U_\theta(s,v)\mid s].
\end{equation}
直接展开可得
\[
D_\theta(s,a)
=\eta\widetilde f_k(s,a)-U_\theta(s,a)
+\mathbb E_{v\sim\pi_k}[U_\theta(s,v)\mid s].
\]
未线性化 BPO 的损失为
\begin{equation}
\label{eq:ch2-bpo-loss}
\mathcal L_{\mathrm{BPO}}(\theta)
=\frac1{2\eta}\mathbb E_{d_k,\pi_k}[D_\theta(s,a)^2].
\end{equation}
由于
\[
\nabla_\theta\mathbb E_{v\sim\pi_k}[U_\theta(s,v)]
=\mathbb E_{v\sim\pi_k}[\nabla_\theta\log\pi_\theta(v\mid s)],
\]
所以
\begin{equation}
\label{eq:ch2-bpo-gradient}
-\eta\nabla_\theta\mathcal L_{\mathrm{BPO}}
=\mathbb E_{d_k,\pi_k}\left[D_\theta(s,a)
\left(\nabla_\theta\log\pi_\theta(a\mid s)
-\mathbb E_{v\sim\pi_k}[\nabla_\theta\log\pi_\theta(v\mid s)\mid s]\right)\right].
\end{equation}
BPO 的 score centering 来自对当前 student--anchor KL 的精确求导；它不是因为先对 reward 做了一个近似。

\subsection{MDPO：直接优化 mirror objective}
\label{sec:ch2-mdpo-generic}

MDPO 直接最大化式 \eqref{eq:ch2-mirror} 的参数化版本：
\begin{equation}
\label{eq:ch2-mdpo-generic-objective}
\mathcal J_{\mathrm{MDPO}}(\theta)
=\mathbb E_{a\sim\pi_\theta}[f(s,a)]
-\frac1\eta\operatorname{KL}\!\left(\pi_\theta(\cdot\mid s)\middle\Vert\pi_k(\cdot\mid s)\right).
\end{equation}
完整词表时 KL 是
\begin{equation}
\label{eq:ch2-mdpo-full-kl}
\operatorname{KL}(\pi_\theta\Vert\pi_k)
=\sum_a\pi_\theta(a\mid s)
\left[\log\pi_\theta(a\mid s)-\log\pi_k(a\mid s)\right].
\end{equation}
若 action 从 anchor 采样，定义 $R_\theta(s,a)=\exp(\ell^\theta(a)-\ell^k(a))$，则 action-level IS 形式为
\begin{equation}
\label{eq:ch2-mdpo-sample-integrand}
\widehat{\mathcal J}_{\mathrm{MDPO}}
=\mathbb E_{a\sim\pi_k}\left[R_\theta(s,a)f(s,a)-\frac1\eta R_\theta(s,a)\log R_\theta(s,a)\right].
\end{equation}
其 ascent direction 为
\begin{equation}
\label{eq:ch2-mdpo-generic-gradient}
\nabla_\theta\mathcal J_{\mathrm{MDPO}}
=\mathbb E_{\pi_k}\left[R_\theta(s,a)\left(f(s,a)-\frac1\eta\log R_\theta(s,a)\right)\nabla_\theta\log\pi_\theta(a\mid s)\right].
\end{equation}
额外的 $-R_\theta/\eta$ 项乘以 score 后，在完整 action 求和中因 $\mathbb E_{\pi_\theta}[\nabla\log\pi_\theta]=0$ 消失。MDPO 因而直接优化 mirror objective，其他三种方法优化的是由同一最优性方程构造的 regression objective。

\begin{theorem}[四种 inner 的共同精确解]
\label{thm:ch2-inner-common-root}
在 tabular policy、完整 action 支持和精确词表期望下，PMD-Part、PMD-Mean、未线性化 BPO 和 MDPO 都以式 \eqref{eq:ch2-optimal-policy} 为精确解。PMD-Mean 的结论要求其投影 target 可实现；参数化 student 或有限样本时，四者得到的是不同的投影或优化误差。
\end{theorem}

\begin{proof}
PMD-Part 在零损失处满足 $U_\theta=\eta f-\log Z_f$，于是得到式 \eqref{eq:ch2-optimal-policy}。BPO 的零 residual 只确定 $U_\theta=\eta f-C(s)$；代入 $\mathbb E_{\pi_k}[e^{U_\theta}]=1$ 得 $C(s)=\log Z_f(s)$。PMD-Mean 若中心化 target 属于归一化约束集合，则其平方投影残差为零，并由中心化不变性得到同一策略。MDPO 由式 \eqref{eq:ch2-kl-decomp} 直接得到唯一最大点。
\end{proof}

\section{OPD：从 teacher log-ratio 到 outer--inner 算法}
\label{sec:ch2-opd-instantiation}

OPD 不引入额外的外部 reward，而在第 $k$ 个 outer step 使用 teacher 与当前 anchor 的局部 log-ratio：
\begin{equation}
\label{eq:ch2-opd-current-reward}
r_k(s,a)=\log T(a\mid s)-\log\pi_k(a\mid s).
\end{equation}
它是序列 log-ratio 的逐 token 分解：
\begin{equation}
\label{eq:ch2-opd-sequence-telescope}
\sum_{t=1}^{H}r_k(s_t,a_t)
=\log T(y\mid x)-\log\pi_k(y\mid x).
\end{equation}
因此 $r_k$ 是对 teacher--student 序列差异的无损分摊，不是一个与 prefix 无关的固定终端 reward。

\begin{definition}[OPD mirror target]
\label{def:ch2-opd-target}
将 $f=r_k$ 代入式 \eqref{eq:ch2-optimal-policy}，定义
\begin{equation}
\label{eq:ch2-opd-geometric-bridge}
\pi_{k+1}^{\star}(a\mid s)
=\frac{\pi_k(a\mid s)^{1-\eta}T(a\mid s)^\eta}{Z_k(s)},
\qquad
Z_k(s)=\sum_v\pi_k(v\mid s)^{1-\eta}T(v\mid s)^\eta.
\end{equation}
\end{definition}

\begin{theorem}[OPD mirror step 的几何收敛]
\label{thm:ch2-opd-geometric-bridge}
若 $0<\eta\leq1$，则精确 outer 更新满足
\begin{equation}
\label{eq:ch2-opd-projective-contraction}
\log\frac{\pi_{k+1}^{\star}(a\mid s)}{T(a\mid s)}
=(1-\eta)\log\frac{\pi_k(a\mid s)}{T(a\mid s)}-\log Z_k(s).
\end{equation}
因此任意两个 action 的 log-ratio 差满足
\[
\max_a\log\frac{\pi_{k+1}^{\star}(a\mid s)}{T(a\mid s)}
-\min_a\log\frac{\pi_{k+1}^{\star}(a\mid s)}{T(a\mid s)}
=(1-\eta)\Delta_k(s).
\]
$\eta=1$ 时一步得到 $T$；$0<\eta<1$ 时 projective 误差按 $1-\eta$ 几何衰减。
\end{theorem}

\begin{proof}
由式 \eqref{eq:ch2-opd-geometric-bridge}，
\[
\log\frac{\pi_{k+1}^{\star}(a\mid s)}{T(a\mid s)}
=(1-\eta)\log\frac{\pi_k(a\mid s)}{T(a\mid s)}-\log Z_k(s).
\]
减去任意两个 action 的等式，归一化常数抵消，得到 projective 递推。
\end{proof}

\section{四种 OPD inner 的可执行估计量}
\label{sec:ch2-opd-four-inner}

对第 $i$ 个训练 prefix $s_i$，采样 $K$ 个 anchor token $y_{ij}\sim\pi_k(\cdot\mid s_i)$，记录
\begin{equation}
\label{eq:ch2-opd-three-logprobs}
\ell^T_{ij}=\log T(y_{ij}\mid s_i),\qquad
\ell^k_{ij}=\log\pi_k(y_{ij}\mid s_i),\qquad
\ell^\theta_{ij}=\log\pi_\theta(y_{ij}\mid s_i).
\end{equation}
完整词表 logits 给出
\begin{equation}
\label{eq:ch2-opd-forward-kls}
D_i^{kT}=\sum_v\pi_k(v\mid s_i)[\log\pi_k(v\mid s_i)-\log T(v\mid s_i)],
\quad
D_i^{k\theta}=\sum_v\pi_k(v\mid s_i)[\log\pi_k(v\mid s_i)-\log\pi_\theta(v\mid s_i)].
\end{equation}
如果没有完整词表求和，可用同一 prefix 的 sampled estimates
\[
\widehat D_i^{kT}=K^{-1}\sum_j(\ell^k_{ij}-\ell^T_{ij}),
\qquad
\widehat D_i^{k\theta}=K^{-1}\sum_j(\ell^k_{ij}-\ell^\theta_{ij}),
\]
并停止中心量的梯度。以下所有式子都可以逐项实现。

\paragraph{PMD-Part。}
\begin{equation}
\label{eq:ch2-opd-part-explicit-loss}
\widehat{\mathcal L}_{k,\mathrm{Part}}
=\frac1{2BK}\sum_{i,j}
\left[\ell^\theta_{ij}-\ell^k_{ij}
-\eta(\ell^T_{ij}-\ell^k_{ij})+\log Z_k(s_i)\right]^2.
\end{equation}
\begin{equation}
\label{eq:ch2-opd-part-explicit-gradient}
-\widehat\nabla\mathcal L_{k,\mathrm{Part}}
=\frac1{BK}\sum_{i,j}
\left[\eta(\ell^T_{ij}-\ell^k_{ij})-\ell^\theta_{ij}+\ell^k_{ij}-\log Z_k(s_i)\right]
\nabla_\theta\log\pi_\theta(y_{ij}\mid s_i).
\end{equation}
\paragraph{PMD-Mean。}
\begin{equation}
\label{eq:ch2-opd-mean-explicit-loss}
\widehat{\mathcal L}_{k,\mathrm{Mean}}
=\frac1{2BK}\sum_{i,j}
\left[\ell^\theta_{ij}-\ell^k_{ij}
-\eta(\ell^T_{ij}-\ell^k_{ij}+D_i^{kT})\right]^2,
\end{equation}
\begin{equation}
\label{eq:ch2-opd-mean-explicit-gradient}
-\widehat\nabla\mathcal L_{k,\mathrm{Mean}}
=\frac1{BK}\sum_{i,j}
\left[\eta(\ell^T_{ij}-\ell^k_{ij}+D_i^{kT})-\ell^\theta_{ij}+\ell^k_{ij}\right]
\nabla_\theta\log\pi_\theta(y_{ij}\mid s_i).
\end{equation}
\paragraph{BPO。}
\begin{equation}
\label{eq:ch2-opd-bpo-explicit-residual}
\widehat\delta_{ij}
=\eta(\ell^T_{ij}-\ell^k_{ij}+D_i^{kT})
-\left(\ell^\theta_{ij}-\ell^k_{ij}+D_i^{k\theta}\right).
\end{equation}
\begin{equation}
\label{eq:ch2-opd-bpo-explicit-loss}
\widehat{\mathcal L}_{k,\mathrm{BPO}}
=\frac1{2\eta BK}\sum_{i,j}\widehat\delta_{ij}^{\,2},
\end{equation}
\begin{equation}
\label{eq:ch2-opd-bpo-explicit-gradient}
-\eta\widehat\nabla\mathcal L_{k,\mathrm{BPO}}
=\frac1{BK}\sum_{i,j}\widehat\delta_{ij}
\left[\nabla_\theta\log\pi_\theta(y_{ij}\mid s_i)
-\sum_v\pi_k(v\mid s_i)\nabla_\theta\log\pi_\theta(v\mid s_i)\right].
\end{equation}
实际 sampled 实现把第二项替换为 $K^{-1}\sum_\ell\nabla_\theta\log\pi_\theta(y_{i\ell}\mid s_i)$，并把 $D_i^{k\theta}$ 替换为停止梯度的 $\widehat D_i^{k\theta}$。
\paragraph{MDPO。}
令 $R_{ij}^{\theta}=\exp(\ell^\theta_{ij}-\ell^k_{ij})$。anchor token 样本上的 objective 与 ascent estimator 分别为
\begin{equation}
\label{eq:ch2-opd-mdpo-explicit-objective}
\widehat{\mathcal J}_{k,\mathrm{MDPO}}
=\frac1{BK}\sum_{i,j}R_{ij}^{\theta}
\left[(\ell^T_{ij}-\ell^k_{ij})-\frac1\eta(\ell^\theta_{ij}-\ell^k_{ij})\right],
\end{equation}
\begin{equation}
\label{eq:ch2-opd-mdpo-explicit-gradient}
\widehat\nabla\mathcal J_{k,\mathrm{MDPO}}
=\frac1{BK}\sum_{i,j}R_{ij}^{\theta}
\left[(\ell^T_{ij}-\ell^k_{ij})-\frac1\eta(\ell^\theta_{ij}-\ell^k_{ij})\right]
\nabla_\theta\log\pi_\theta(y_{ij}\mid s_i).
\end{equation}

\begin{proposition}[PMD-Mean、BPO 与 PMD-Part 的差异]
\label{prop:ch2-mean-bpo-equivalence}
三者共享式 \eqref{eq:ch2-optimal-policy} 的精确零点，但有限步梯度一般不同。PMD-Mean 使用 $\eta\widetilde f_k-U_\theta$ 乘以 ordinary student score；BPO 使用
\[
D_\theta=\eta\widetilde f_k-U_\theta+\mathbb E_{\pi_k}[U_\theta]
\]
乘以 centered student score。二者只有在
\[
\mathbb E_{\pi_k}[U_\theta]=0,
\qquad
\mathbb E_{\pi_k}[\nabla\log\pi_\theta]=0
\]
同时成立时才给出相同梯度。
\end{proposition}

\begin{proof}
将 $D_\theta$ 与 PMD-Mean residual 相减得到
\[
D_\theta-(\eta\widetilde f_k-U_\theta)=\mathbb E_{\pi_k}[U_\theta].
\]
BPO 的 score 由 $D_\theta$ 对 student 参数求导产生，正好是 ordinary score 减去其 anchor 均值。代入两式并展开，差异由上述两个 action-independent 项的乘积给出；它们并非由策略归一化自动消失，因为归一化只保证 $\mathbb E_{\pi_\theta}[\nabla\log\pi_\theta]=0$。
\end{proof}

\section{Inner 误差、outer 更新与 Natural Gradient}
\label{sec:ch2-inner-outer}

一个 outer step 先冻结 $\pi_k$ 和 teacher $T$，再用上面的任一 inner objective 更新 student 参数 $M$ 次，最后令 $\pi_{k+1}=\pi_{\theta_M}$。$M$ 是优化预算，不等于数学上的精确收敛；精确解是式 \eqref{eq:ch2-opd-geometric-bridge}。

\begin{proposition}[有限 inner 误差的 outer 递推]
\label{prop:ch2-opd-inner-error}
令 $\Delta_k(s)$ 为式 \eqref{eq:ch2-opd-projective-contraction} 左侧的 projective 振幅。若第 $k$ 步 inner 产生的 student 满足
\[
\Delta_{k+1}(s)\leq(1-\eta)\Delta_k(s)+\varepsilon_k(s),
\]
则
\[
\Delta_{k}(s)\leq(1-\eta)^k\Delta_0(s)
+\sum_{j=0}^{k-1}(1-\eta)^j\varepsilon_{k-1-j}(s).
\]
若 $\sup_k\varepsilon_k(s)\leq\bar\varepsilon(s)$，则
\[
\limsup_{k\to\infty}\Delta_k(s)\leq\bar\varepsilon(s)/\eta.
\]
\end{proposition}

\begin{proof}
对递推式反复代入即可得到有限 $k$ 的展开。几何级数满足 $\sum_{j\geq0}(1-\eta)^j=1/\eta$，取上极限得到结论。
\end{proof}

在参数空间，mirror objective 的 KL 项在 $\theta_k$ 附近具有 Fisher 曲率。若 $z_\theta(s)$ 是 student logits，$F_k(s)=\operatorname{Diag}(\pi_k)-\pi_k\pi_k^{\mathsf T}$，则
\[
\operatorname{KL}(\pi_\theta\Vert\pi_k)
=\frac12\Delta z^{\mathsf T}F_k\Delta z+O(\|\Delta z\|^3).
\]
因此在信赖域足够小且使用 Fisher 预条件时，mirror outer step 的一阶参数更新是 natural-gradient step；普通 SGD 只沿欧氏坐标梯度移动，步长会随参数化和分布熵改变，而 natural gradient 按局部 KL 几何校准。这里的等价性是局部一阶等价，不意味着任意 AdamW 训练都等于精确 mirror update。

\section{访问测度的最小必要说明}
\label{sec:ch2-access-measure}

上面的 inner 公式默认 prefix 按 $d_{\pi_k}$ 采样。若训练数据来自另一个 replay prefix measure $\nu$，定义 Radon--Nikodym 权重
\[
\Omega_k(s)=\frac{d_{\pi_k}(s)}{d_{\nu}(s)}.
\]
对任何只依赖当前 $(s,a)$ 的 inner integrand $Y(s,a)$，有
\begin{equation}
\label{eq:ch2-opd-prefix-is}
\mathbb E_{s\sim d_{\pi_k},\,a\sim\pi_k}[Y(s,a)]
=\mathbb E_{s\sim d_\nu,\,a\sim\pi_k}[\Omega_k(s)Y(s,a)].
\end{equation}
这说明 prefix 权重只负责把访问测度从 replay 移回 anchor；它不改变四种 inner 的 token-level 公式。对于局部 OPD reward，不需要额外乘后缀权重；只有当目标改成终端序列 return 时，才应重新推导完整的 sequence importance sampling。

\begin{remark}
本章把 replay 的访问信息统一写成目标 prefix 测度 $d_{\pi_k}$、数据 prefix 测度 $d_\nu$ 及其比值 $\Omega_k$；该比值只修正 prefix 访问测度，与 MDPO 的 action ratio 分开定义。
\end{remark}

\chapter{梯度的偏差与方差}
\label{ch:bias-variance}

\section{估计对象必须先固定}

固定 prefix $s$，令 $g(a)=\nabla_\theta\log\cand_\theta(a\mid s)$、$u(a)=\log(\cand_\theta(a\mid s)/\beh(a\mid s))$，并令 $\rho(a)=e^{u(a)}$。以下区分三个不同的随机量：

\begin{enumerate}[label=(\roman*)]
    \item 直接 KL 镜像目标的梯度 $\nabla F_s$；
    \item 行为测度下回归损失的梯度 $-\nabla\mathcal L_{\rm Part}$；
    \item BPO 的逐 prefix 残差损失梯度。
\end{enumerate}

三者在精确镜像解处共享零点，但只有第一个是 $F_s$ 的梯度。偏差必须相对于明确的目标定义；把行为测度下的回归梯度称为 ``无偏 off-policy policy gradient'' 会混淆两个目标。

\section{MDPO：无偏但承担 action-ratio 的二阶矩}

令 $c(a)=f(s,a)-\frac1\eta u(a)$。MDPO 的单样本估计量为
\begin{equation}
\widehat G_{\rm MDPO}=\rho(a)c(a)g(a),\qquad a\sim\beh(\cdot\mid s),
\end{equation}
并且在 $\cand\ll\beh$、$\E_\beh[\rho\lVert cg\rVert]<\infty$ 时
\begin{equation}
\E_\beh[\widehat G_{\rm MDPO}]=\E_\cand[cg]=\nabla F_s.
\end{equation}
因此 action-level IS 本身不产生偏差；若 prefix 来自 $d_\beh$ 而目标要求 $d_\cand$，还必须乘 $W_{<t}=d_\cand(s_t)/d_\beh(s_t)$。这两个密度比修正的是不同测度。

\begin{theorem}[MDPO 梯度的精确偏差--方差分解]
\label{thm:mdpo-exact-mse}
设 $G=\E_\beh[\rho cg]$，批量大小为 $n$，并假设 $\E_\beh[\rho^2c^2\lVert g\rVert^2]<\infty$。则样本均值 $\widehat G_n$ 满足
\begin{align}
\operatorname{Bias}(\widehat G_n)&=0,\\
\E\lVert\widehat G_n-G\rVert^2
&=\frac1n\left(\E_\beh[\rho^2c^2\lVert g\rVert^2]-\lVert G\rVert^2\right).
\end{align}
若 $\lVert g\rVert\leq B_g$、$|c|\leq B_c$，则
\begin{equation}
\operatorname{tr}\operatorname{Cov}(\widehat G_n)
\leq\frac{B_g^2B_c^2}{n}(1+\chi^2(\cand\Vert\beh)).
\end{equation}
\end{theorem}

\begin{proof}
令
\[
X_i=\rho(a_i)c(a_i)g(a_i),
\qquad a_i\stackrel{\mathrm{i.i.d.}}{\sim}\beh,
\qquad \widehat G_n=\frac1n\sum_{i=1}^nX_i.
\]
由 support 条件，$\rho=\cand/\beh$ 在 $\beh$-几乎处处有定义。换测度恒等式给出
\begin{align*}
\E_\beh[X_i]
&=\sum_a\beh(a)\frac{\cand(a)}{\beh(a)}c(a)g(a)\\
&=\sum_a\cand(a)c(a)g(a)=\E_\cand[cg]=G.
\end{align*}
因此
$\E[\widehat G_n]=n^{-1}\sum_i\E[X_i]=G$，样本均值的偏差严格为零。
再令 $Y_i=X_i-G$。独立性和 $\E[Y_i]=0$ 消掉交叉项，故
\begin{align*}
\E\lVert\widehat G_n-G\rVert^2
&=\E\left\lVert\frac1n\sum_iY_i\right\rVert^2\\
&=\frac{1}{n^2}\sum_i\E\lVert Y_i\rVert^2
=\frac1n\left(\E\lVert X_1\rVert^2-\lVert G\rVert^2\right).
\end{align*}
这就是向量均方误差公式。最后，若 $\lVert g\rVert\leq B_g$ 且 $|c|\leq B_c$，则
\[
\E_\beh\lVert X_1\rVert^2
\leq B_g^2B_c^2\E_\beh[\rho^2]
=B_g^2B_c^2\bigl(1+\chi^2(\cand\Vert\beh)\bigr),
\]
其中最后一个等号来自
$\chi^2(\cand\Vert\beh)=\E_\beh[(\rho-1)^2]=\E_\beh[\rho^2]-1$。去掉非负项 $\lVert G\rVert^2$ 即得到协方差迹的上界。
\end{proof}

在自回归序列中，$W_{\leq t}=\prod_{k\leq t}\rho_k$ 的二阶矩可以逐步展开。给定 prefix $y_{<t}$，有
\[
\E_\beh[\rho_t^2\mid y_{<t}]
=\sum_a\beh(a\mid y_{<t})
\left(\frac{\cand(a\mid y_{<t})}{\beh(a\mid y_{<t})}\right)^2
=1+\chi^2\bigl(\cand(\cdot\mid y_{<t})\Vert\beh(\cdot\mid y_{<t})\bigr).
\]
因为 $W_{\leq t}=W_{<t}\rho_t$，塔式法则给出
\begin{align*}
\E_\beh[W_{\leq t}^2]
&=\E_\beh\left[W_{<t}^2\E_\beh[\rho_t^2\mid y_{<t}]\right]\\
&=\E_\beh\left[W_{<t}^2\left(1+\chi^2(s_t)\right)\right].
\end{align*}
若所有可达 prefix 都满足 $c_{\rm lo}\leq\chi^2(s)\leq c_{\rm hi}$，则递推归纳得到
\begin{equation}
(1+c_{\rm lo})^t\leq\E_\beh[W_{\leq t}^2]\leq(1+c_{\rm hi})^t.
\end{equation}
这给出了 prefix IS 方差随修正长度增长的明确阶数。

再考虑完整权重 $W=W_{\leq t}W_{>t}$。在条件 $y_{\leq t}$ 下，后缀权重的均值为一：$\E[W_{>t}\mid y_{\leq t}]=1$，因此
$\E[W\mid y_{\leq t}]=W_{\leq t}$。全方差公式于是给出
\begin{equation}
\label{eq:prefix-total-variance}
\begin{aligned}
\operatorname{Var}(W)
&=\operatorname{Var}(\E[W\mid y_{\leq t}])
 +\E[\operatorname{Var}(W\mid y_{\leq t})]\\
&=\operatorname{Var}(W_{\leq t})
 +\E[W_{\leq t}^2\operatorname{Var}(W_{>t}\mid y_{\leq t})].
\end{aligned}
\end{equation}
第二项非负；只要 prefix 之后存在策略比率不恒等于一的可达状态，它就严格为正。这说明后缀权重只能增加方差，不能抵消 prefix 权重已经产生的方差。

\section{PMD-Part 的投影偏差与 BPO 的中心化方差}

设策略类受限为 $\Pi$。PMD-Part 的 population 解为
\begin{equation}
\theta_{\rm Part}^*=\argmin_{\theta\in\Pi}\E_{d_\beh,\beh}[(u_\theta-h)^2],
\end{equation}
而 BPO 的逐 prefix 解为
\begin{equation}
\theta_{\rm BPO}^*=\argmin_{\theta\in\Pi}\E_{d_\beh}\operatorname{Var}_{a\sim\beh}[u_\theta(a)-\eta f(a)].
\end{equation}
二者在可实现时拥有同一零点；不可实现时，BPO 不惩罚 action-independent 的误差，因此它允许更大的常数方向自由度，而 PMD-Part 通过精确的 $\log Z_f$ 固定该常数。共享参数网络下，$d_\beh$ 同时决定状态间投影误差如何分配。

\begin{proposition}[prefix IS 的偏差--方差阶梯]
\label{prop:prefix-bias-variance}
对终端或未来回报 $R$，行为样本下的 token-level 不加前缀权重估计量，其期望与目标策略梯度的差可分解为
\begin{equation}
\sum_s(d_\cand-d_\beh)(s)\sum_a\nabla\cand(a\mid s)A^\beh(s,a)
+\sum_s d_\cand(s)\sum_a\nabla\cand(a\mid s)(A^\cand-A^\beh)(s,a).
\end{equation}
乘 $W_{<t}$ 后第一项恰好消失；再乘后缀 $W_{>t}$ 才能修正第二项，但其额外方差正是式 \eqref{eq:prefix-total-variance} 的第二项。
\end{proposition}

\begin{proof}
第一项来自把 $d_\cand$ 替换为 $d_\beh$，第二项来自把 $A^\cand$ 替换为可估计的 $A^\beh$。前缀密度比将行为 prefix 测度换成 $d_\cand$，所以只消掉第一项；后缀比率不改变已经使用的优势估计，却引入条件二阶矩，故只能修正价值滞后而不能降低方差。
\end{proof}

\begin{remark}[可加残差为何能绕开乘积]
若信号写成逐 token 鞅差之和 $D=\sum_tD_t$，且 $\E[D_t\mid s_t]=0$，则交叉项消失并有
\begin{equation}
\E[D^2]=\sum_t\E[D_t^2].
\end{equation}
这正是离策略修正文档中 ``可加泛函的换测度界'' 的核心：回归型 PMD/BPO 利用逐点最优性或中心化残差，把序列乘积权重替换成逐步二阶量；代价是目标从真策略梯度改成行为测度下的投影问题。

\end{remark}

\begin{proposition}[可加 BPO 残差的损失估计方差]
\label{prop:bpo-loss-variance}
设轨迹残差可写成 $\delta=\sum_{t=1}^{T}e_t$，其中 $e_t$ 是关于 prefix 过滤的鞅差，且 $|e_t|\leq B_e$。则
\begin{equation}
\E[\delta^2]=\sum_{t=1}^{T}\E[e_t^2],
\qquad
\E[\delta^4]\leq T^2B_e^2\sum_{t=1}^{T}\E[e_t^2].
\end{equation}
对 $n$ 条独立轨迹的平方残差损失估计量
$\widehat{\mathcal L}=n^{-1}\sum_i\delta_i^2/(2\eta)$，有
\begin{equation}
\operatorname{Var}(\widehat{\mathcal L})
\leq \frac{T^2B_e^2}{4\eta^2n}
\sum_{t=1}^{T}\E[e_t^2].
\end{equation}
因此中心化残差的统计方差按 token 二阶量相加，而不是按 sequence ratio 的二阶矩相乘。
\end{proposition}

\begin{proof}
鞅差正交给出第一式。由 Cauchy--Schwarz，
$\delta^2=(\sum_te_t)^2\leq T\sum_te_t^2\leq T^2B_e^2$，故
$\delta^4\leq T^2B_e^2\delta^2$。独立样本均值的方差为单样本方差除以 $n$，再用
$\operatorname{Var}(\delta^2)\leq\E[\delta^4]$ 即得结论。
\end{proof}

\section{BPO 论文的训练预算}
\label{sec:bpo-implementation}

BPO 论文源码的主实验写明：每个 training step 先生成 256 个 prompt、每个 prompt 16 个 response，共 4096 条 response；再划分为 8 个、每个 512 条 response 的 mini-batch，执行 8 次 optimizer updates。主实验运行 400 个 training steps，即 3200 次 optimizer updates；附录消融运行 1000 个 training steps。论文没有按 residual norm 或 mirror-subproblem gap 停止的 inner loop，因此这些是固定更新预算，不是子问题收敛步数。

\section{OpenKimi/PMD 的训练预算}
\label{sec:openkimi-implementation}

OpenKimi 的 PMD loss 默认先对完整 response 的 token log-ratio 求和，再把 sequence log-ratio broadcast 到 response 内 token。\par Qwen2.5-7B math recipe 设置 `PPO\_EPOCHS=1`、`train\_batch\_size=512`、`actor.ppo\_mini\_batch\_size=32`，所以静态 batch 下每个 outer step 约有 $512/32=16$ 个 mini-batch 更新。代码没有 inner residual stopping 或 PMD gap 检验；`total\_training\_steps=3000` 是 outer steps 上限。

\section{经典 MDPO 的训练预算与 IS}
\label{sec:mdpo-implementation}

经典 MDPO on-policy 代码默认 `sgd\_steps=5`，MuJoCo README 的推荐示例使用 10。一个 rollout batch 固定后，对同一批数据重复执行该数量的 policy optimizer updates，再刷新 old/new anchor；没有根据 mirror residual 自适应停止。

off-policy 代码的 README 暴露 `sgd\_steps=1000`，但当前训练循环实际写为 `for grad\_step in range(1)`，原本的 `int(self.gradient\_steps)` 被注释。因此每个 train event 实际只执行一次 actor/critic update；`gradient\_steps` 主要影响 target/anchor policy 同步周期。实现使用 replay 状态测度和 action-level policy gradient，不能解释为完整的 prefix-IS 求解。

\section{比较时应报告的指标}
\label{sec:reporting}

为了比较四种方法，实验至少应固定同一 rollout batch，并记录：每个 inner update 的 mirror loss、log-ratio residual norm、$\KL(\cand_\theta\Vert\beh)$、梯度范数、ratio 的均值与二阶矩、中心化 residual 的条件方差，以及 prefix coverage。这样才能区分“做了更多 optimizer steps”和“真正接近 token-level mirror optimum”。

\chapter{结论与 MirrorOPD 的边界}
\label{ch:conclusion}

从同一个 token-level Gibbs 最优解出发，PMD-Part 是精确 log-ratio 回归，PMD-Mean 是中心化回归后的 Lambert-$W$ 投影，MDPO 是带 action IS 的直接 KL 正则化 policy gradient。BPO 的完整 reverse-KL 平方 residual 在未线性化时产生中心化 score 梯度；它与 PMD-Part 共享可实现时的最优点，但一般相差 $c_s\bar g$，因此不能把二者的梯度直接视为相同。线性化、二元 KL 化简和工程截断属于后续近似，本文件没有把它们作为理论等价的前提。

本文件暂不推导 OPD 的完整 off-policy correction 或 KL 系数调度。后续 MirrorOPD 需要先选择局部即时 reward 目标还是完整加性回报目标，再决定是否引入 prefix IS、是否采用 Part/Mean 投影，以及如何用 Natural Gradient 控制学生策略与教师策略的距离。
