\chapter{统一的 token-level Mirror Descent 与 OPD 实例化}
\label{ch:four-methods}

本章固定一套与 outer--inner 动力学一致的记号。给定 prompt $x$，rollout 为
$y_{1:H}$，第 $t$ 个 token 之前的历史为 $s_t=(x,y_{<t})$，词表为
$\mathcal V$。第 $k$ 个 outer step 的冻结 anchor policy 记为 $\pi_k$，该轮
第 $n$ 个 inner step 的 student policy 记为
\begin{equation}
\pi_{k,n}(a\mid s)\coloneqq\pi_{\theta_{k,n}}(a\mid s),
\qquad \pi_{k,0}=\pi_k .
\end{equation}
teacher policy 记为 $\pi_T$；它在一个 outer step 内始终冻结。镜像系数记为
$\eta>0$，inner optimizer 的步长记为 $\alpha_{k,n}$，两者不能混用。

为避免把不同方向的 KL 和 top-$K$ 中的整数 $K$ 混在一起，本章使用
\begin{align}
\Delta_{k,n}(s,a)&\coloneqq
\log\frac{\pi_{k,n}(a\mid s)}{\pi_k(a\mid s)},\\
r_k(s,a)&\coloneqq
\log\frac{\pi_T(a\mid s)}{\pi_k(a\mid s)},\\
K_k^T(s)&\coloneqq
\operatorname{KL}\!\left(\pi_k(\cdot\mid s)\middle\Vert\pi_T(\cdot\mid s)\right),\\
K_{k,n}^\theta(s)&\coloneqq
\operatorname{KL}\!\left(\pi_k(\cdot\mid s)\middle\Vert\pi_{k,n}(\cdot\mid s)\right).
\end{align}
因此
$\mathbb E_{\pi_k}[r_k\mid s]=-K_k^T(s)$，而
$\mathbb E_{\pi_k}[\Delta_{k,n}\mid s]=-K_{k,n}^\theta(s)$。
这些等式解释了后文两个 KL 的方向：它们都是在冻结 anchor 的 action 测度下
取均值，而不是 MDPO 中以 student 为左侧分布的 reverse KL。$d_k$ 表示由
$\pi_k$ rollout 诱导的 prefix 访问测度；所有 inner 目标先对
$s\sim d_k$、再对 $a\sim\pi_k(\cdot\mid s)$ 取期望。

\section{共同的 mirror 子问题与精确解}
\label{sec:ch2-common}

\begin{definition}[Token-level mirror 子问题]
\label{def:ch2-mirror-subproblem}
给定固定历史 $s$、token 信号 $f(s,a)$ 和 anchor $\pi_k$，定义
\begin{equation}
\label{eq:ch2-mirror}
\mathcal M_s(p;f,\pi_k)
=\mathbb E_{a\sim p(\cdot\mid s)}[f(s,a)]
-\frac1\eta\operatorname{KL}\!\left(p(\cdot\mid s)\middle\Vert\pi_k(\cdot\mid s)\right).
\end{equation}
镜像子问题是在所有满足 $p\ll\pi_k$ 的概率分布上最大化 $\mathcal M_s$。同时定义配分函数
\begin{equation}
\label{eq:ch2-mirror-normalizer}
Z_f(s)=\sum_{v\in\mathcal V}\pi_k(v\mid s)\exp\!\bigl(\eta f(s,v)\bigr).
\end{equation}
\end{definition}

\begin{theorem}[KL mirror 子问题的 Gibbs 解]
\label{thm:ch2-gibbs}
在有限词表、$Z_f(s)<\infty$ 且 $\pi_k(v\mid s)>0$ 的条件下，式 \eqref{eq:ch2-mirror} 的唯一解是
\begin{equation}
\label{eq:ch2-optimal-policy}
\pi_k^+(v\mid s)=\frac{\pi_k(v\mid s)e^{\eta f(s,v)}}{Z_f(s)},
\qquad
\log\frac{\pi_k^+(v\mid s)}{\pi_k(v\mid s)}
=\eta f(s,v)-\log Z_f(s).
\end{equation}
对任意候选 $p$，还有精确分解
\begin{equation}
\label{eq:ch2-kl-decomp}
\mathcal M_s(p;f,\pi_k)
=\frac1\eta\log Z_f(s)
-\frac1\eta\operatorname{KL}\!\left(p\middle\Vert\pi_k^+(\cdot\mid s)\right).
\end{equation}
\end{theorem}

\begin{proof}
由式 \eqref{eq:ch2-optimal-policy}，有
\[
\log\frac{p(v\mid s)}{\pi_k^+(v\mid s)}
=\log\frac{p(v\mid s)}{\pi_k(v\mid s)}-\eta f(s,v)+\log Z_f(s).
\]
对 $v$ 按 $p$ 求期望，得到
\[
\operatorname{KL}(p\Vert\pi_k^+)
=\operatorname{KL}(p\Vert\pi_k)-\eta\mathbb E_p[f]+\log Z_f(s).
\]
移项即得式 \eqref{eq:ch2-kl-decomp}。KL 非负且仅在 $p=\pi_k^+$ 时取零，因此最优解唯一。
\end{proof}

\begin{definition}[中心化 token 信号]
\label{def:ch2-centered-signal}
对任意 $f$ 定义 anchor 条件均值和中心化信号
\[
\bar f_k(s)=\mathbb E_{a\sim\pi_k}[f(s,a)\mid s],
\qquad
\widetilde f_k(s,a)=f(s,a)-\bar f_k(s).
\]
若 $f=A^{\pi_k}$ 是 anchor 的优势函数，则 $\bar f_k(s)=0$ 是优势函数定义推出的恒等式；对一般 teacher score 或 OPD log-ratio，$\bar f_k(s)=0$ 不是自动成立的假设。
\end{definition}

\begin{proposition}[优势函数的 anchor 条件均值]
\label{prop:ch2-advantage-centering}
若 $f(s,a)=A^{\pi_k}(s,a)$，则对每个满足 $\pi_k(\cdot\mid s)$ 定义的历史，
\begin{equation}
\label{eq:ch2-advantage-zero-mean}
\mathbb E_{a\sim\pi_k(\cdot\mid s)}[f(s,a)\mid s]=0.
\end{equation}
对于一般 OPD log-ratio 或 teacher score，该等式不是定义恒等式，必须显式估计并减去条件均值。
\end{proposition}

\begin{proof}
由 $A^{\pi_k}(s,a)=Q^{\pi_k}(s,a)-V^{\pi_k}(s)$ 以及
$V^{\pi_k}(s)=\mathbb E_{a\sim\pi_k}[Q^{\pi_k}(s,a)\mid s]$，有
\begin{align*}
\mathbb E_{\pi_k}[A^{\pi_k}(s,a)\mid s]
&=\mathbb E_{\pi_k}[Q^{\pi_k}(s,a)\mid s]-V^{\pi_k}(s)\\
&=V^{\pi_k}(s)-V^{\pi_k}(s)=0.
\end{align*}
\end{proof}

\begin{proposition}[中心化不改变 mirror 最优策略]
\label{prop:ch2-centering}
用 $\widetilde f_k$ 替代 $f$ 不改变式 \eqref{eq:ch2-mirror} 的最优分布。
\end{proposition}

\begin{proof}
因为 $\bar f_k(s)$ 与 action 无关，
\[
\pi_k(v\mid s)e^{\eta\widetilde f_k(s,v)}
=e^{-\eta\bar f_k(s)}\pi_k(v\mid s)e^{\eta f(s,v)}.
\]
分子和归一化常数同时乘以同一个因子 $e^{-\eta\bar f_k(s)}$，归一化后的分布不变。
\end{proof}

\section{从镜像最优性方程到四种可训练目标}
\label{sec:ch2-objective-derivation}

定理 \ref{thm:ch2-gibbs} 给出的不是孤立的分布结论，而是一条可以被 inner
optimizer 直接拟合的逐 token 方程。下面先把这条方程、平方损失和 BPO 的
中心化残差逐步写出，再进入四种算法的实现。

\subsection{镜像最优性方程与训练测度}
\label{subsec:ch2-optimality-bridge}

把式 \eqref{eq:ch2-optimal-policy} 移项，得到
\begin{equation}
\label{eq:ch2-logratio-equation}
\log\frac{\pi_k^+(a\mid s)}{\pi_k(a\mid s)}
=\eta f(s,a)-\log Z_f(s).
\end{equation}
在参数化 inner 中，左侧由 $\pi_{k,n}$ 近似。因此定义镜像方程残差
\begin{equation}
\mathscr E_{k,n}(s,a)
\coloneqq
\log\frac{\pi_{k,n}(a\mid s)}{\pi_k(a\mid s)}
-\eta f(s,a)+\log Z_f(s).
\label{eq:ch2-mirror-equation-residual}
\end{equation}
精确 mirror 解的条件是 $\mathscr E_{k,n}(s,a)=0$ 对所有有正概率的
$(s,a)$ 成立。

为什么训练测度是 $\pi_k$ 而不是 $\pi_{k,n}$？在第 $k$ 个 outer step，
rollout 已由冻结的 $\pi_k$ 产生；我们要解决的是固定 anchor 的 mirror
子问题，而不是在 inner 过程中不断改变积分测度。于是参数化投影定义为
\begin{equation}
\lVert \mathscr E_{k,n}\rVert_k^2
\coloneqq
\mathbb E_{s\sim d_k}\mathbb E_{a\sim\pi_k(\cdot\mid s)}
\left[\mathscr E_{k,n}(s,a)^2\right].
\label{eq:ch2-anchor-projection-norm}
\end{equation}
这也是 on-policy rollout 下可直接估计的投影几何。若改用行为测度 $\beta_k$
采样，必须乘以冻结的 importance weight
$\pi_k(a\mid s)/\beta_k(a\mid s)$，否则求解的是 $L_2(d_k\beta_k)$ 投影。

\subsection{PMD-Part 与 PMD-Mean：从方程到平方损失}
\label{subsec:ch2-pmd-loss-derivation}

PMD-Part 直接最小化式 \eqref{eq:ch2-mirror-equation-residual} 的平方。将
student log-ratio 和冻结 target 展开，得到
\begin{equation}
\label{eq:ch2-regression-from-mirror}
\mathcal L_{\mathrm{Part}}(\theta)
=\frac12\mathbb E_{s\sim d_k,a\sim\pi_k}
\left[
\log\frac{\pi_\theta(a\mid s)}{\pi_k(a\mid s)}
-\eta f(s,a)+\operatorname{sg}\!\left(\log Z_f(s)\right)
\right]^2.
\end{equation}
其中 $\operatorname{sg}$ 表示停止梯度；$\pi_k$、$f$ 和 $Z_f$ 在本轮
inner 中都冻结。因为
\[
\nabla_\theta\log\frac{\pi_\theta(a\mid s)}{\pi_k(a\mid s)}
=\nabla_\theta\log\pi_\theta(a\mid s),
\]
其梯度为
\begin{equation}
\nabla_\theta\mathcal L_{\mathrm{Part}}
=\mathbb E_{d_k,\pi_k}
\left[\operatorname{sg}\!\left(
\log\frac{\pi_\theta}{\pi_k}-\eta f+\log Z_f
\right)\nabla_\theta\log\pi_\theta(a\mid s)\right].
\label{eq:ch2-part-gradient-bridge}
\end{equation}
这说明 PMD-Part 是 anchor 测度下对完整 mirror log-ratio 的 $L_2$ 投影，
而不是把一个 teacher label 当作普通的 supervised token label。

PMD-Mean 使用原论文的均值近似，而不是对 student log-ratio 做条件中心化。令
\[
\bar f_k(s)=\mathbb E_{\pi_k}[f(s,\cdot)\mid s],\qquad
G_k(s,a)=\eta\bigl(f(s,a)-\bar f_k(s)\bigr).
\]
将完整 target 中的 $\log Z_f(s)$ 用 $\eta\bar f_k(s)$ 近似，得到
\begin{equation}
\mathcal L_{\mathrm{Mean}}(\theta)
=\frac12\mathbb E_{d_k,\pi_k}
\left[\log\frac{\pi_\theta(a\mid s)}{\pi_k(a\mid s)}
-\eta\bigl(f(s,a)-\bar f_k(s)\bigr)\right]^2.
\label{eq:ch2-mean-loss-bridge}
\end{equation}
这里的 reward center 在整个 inner loop 冻结；损失中没有
$\operatorname{KL}(\pi_k\Vert\pi_\theta)$。以 $U_\theta=\log(\pi_\theta/\pi_k)$ 表示 student log-ratio，直接求导得到
\begin{equation}
\nabla_\theta\mathcal L_{\mathrm{Mean}}
=\mathbb E_{d_k,\pi_k}
\left[(U_\theta-G_k)\nabla_\theta\log\pi_\theta(a\mid s)\right].
\label{eq:ch2-mean-gradient-bridge}
\end{equation}
均值近似的误差为
\[
\delta_f(s)=\log Z_f(s)-\eta\bar f_k(s)
=\log\mathbb E_{\pi_k}[e^{G_k(s,\cdot)}\mid s]\ge0.
\]
非恒定信号对应 $\delta_f(s)>0$，此时 $G_k$ 不满足指数归一化约束。
因此 PMD-Mean 即使具有完整函数类和无限样本，也不恢复原始 Gibbs 解；其受约束总体解见定理~\ref{thm:ch2-mean-lambert}。
若改为拟合 $U_\theta-\mathbb E_{\pi_k}[U_\theta\mid s]-G_k$，得到的是后面的 BPO 中心化残差，已经改变了 PMD-Mean 的目标。

\subsection{BPO：完整展开的中心化最优性残差}
\label{subsec:ch2-bpo-loss-derivation}

BPO 对式 \eqref{eq:ch2-logratio-equation} 的 residual 做条件中心化。定义
\begin{align}
\label{eq:ch2-bpo-from-optimality}
D_{k,n}(s,a)
&\coloneqq\eta f(s,a)
-\log\frac{\pi_{k,n}(a\mid s)}{\pi_k(a\mid s)}\\
&\quad-\mathbb E_{v\sim\pi_k(\cdot\mid s)}
\left[\eta f(s,v)-\log\frac{\pi_{k,n}(v\mid s)}{\pi_k(v\mid s)}\right].
\end{align}
将两个条件均值分别写成 KL，得到
\begin{equation}
\boxed{
D_{k,n}(s,a)=
\eta\left(f(s,a)-\bar f_k(s)\right)
-\log\frac{\pi_{k,n}(a\mid s)}{\pi_k(a\mid s)}
-K_{k,n}^\theta(s).}
\label{eq:ch2-bpo-expanded}
\end{equation}
因为
\[
\mathbb E_{\pi_k}\log\frac{\pi_{k,n}}{\pi_k}
=-\operatorname{KL}(\pi_k\Vert\pi_{k,n})=-K_{k,n}^\theta,
\]
式 \eqref{eq:ch2-bpo-expanded} 中最后一项的符号是负号。在 OPD
$f=r_k=\log(\pi_T/\pi_k)$ 时，$\bar f_k=-K_k^T$，于是
\begin{equation}
D_{k,n}(s,a)=
\eta\left[
\log\frac{\pi_T(a\mid s)}{\pi_k(a\mid s)}+K_k^T(s)
\right]
-\left[
\log\frac{\pi_{k,n}(a\mid s)}{\pi_k(a\mid s)}+K_{k,n}^\theta(s)
\right].
\label{eq:ch2-bpo-opd-expanded}
\end{equation}
这就是后文使用的 BPO 权重，不再把它藏在 $f_k$ 或 $U_\theta$ 中。

BPO 的平方损失为
\begin{equation}
\mathcal L_{\mathrm{BPO}}(\theta)
=\frac1{2\eta}\mathbb E_{d_k,\pi_k}[D_{k,n}(s,a)^2].
\label{eq:ch2-bpo-loss-bridge}
\end{equation}
对 student 求导时，$K_{k,n}^\theta$ 也要微分：
\begin{align}
\nabla_\theta K_{k,n}^\theta(s)
&=-\mathbb E_{v\sim\pi_k}
\left[\nabla_\theta\log\pi_{k,n}(v\mid s)\right],\\
\nabla_\theta D_{k,n}(s,a)
&=-\nabla_\theta\log\pi_{k,n}(a\mid s)
+\mathbb E_{v\sim\pi_k}
\left[\nabla_\theta\log\pi_{k,n}(v\mid s)\right].
\end{align}
因此
\begin{equation}
-\eta\nabla_\theta\mathcal L_{\mathrm{BPO}}
=\mathbb E_{d_k,\pi_k}
\left[D_{k,n}(s,a)\left(
\nabla_\theta\log\pi_{k,n}(a\mid s)
-\mathbb E_{v\sim\pi_k}\nabla_\theta\log\pi_{k,n}(v\mid s)
\right)\right].
\label{eq:ch2-bpo-gradient-bridge}
\end{equation}
其中 $-\eta\nabla\mathcal L_{\mathrm{BPO}}$ 是缩放后的下降方向；不改变 optimizer 步长时，实际下降方向为
\[
-\nabla\mathcal L_{\mathrm{BPO}}
=\mathbb E_{d_k,\pi_k}\left[
\left(f-\bar f_k-\frac{\Delta_{k,n}+K_{k,n}^\theta}{\eta}\right)
\left(g_{k,n}-\bar g_{k,n}\right)\right],
\quad g_{k,n}=\nabla\log\pi_{k,n},\quad
\bar g_{k,n}=\mathbb E_{\pi_k}[g_{k,n}\mid s].
\]
由于 $\mathbb E_{\pi_k}[D_{k,n}\mid s]=0$，第二项在完整词表求和时严格为
零；有限 sampled-token batch 中不能把
$\mathbb E_{\pi_k}[\nabla\log\pi_{k,n}]$ 直接当作零。

\subsection{MDPO：直接最大化原始 mirror objective}
\label{subsec:ch2-mdpo-loss-derivation}

MDPO 不把式 \eqref{eq:ch2-logratio-equation} 改写成平方回归，而直接最大化
\begin{equation}
\mathcal J_{\mathrm{MDPO}}(\theta)
=\mathbb E_{s\sim d_k}\left[
\mathbb E_{a\sim\pi_\theta(\cdot\mid s)}f(s,a)
-\frac1\eta\operatorname{KL}\!\left(\pi_\theta(\cdot\mid s)\middle\Vert\pi_k(\cdot\mid s)\right)
\right].
\label{eq:ch2-mdpo-objective-bridge}
\end{equation}
逐历史地使用式 \eqref{eq:ch2-kl-decomp}，有
\begin{equation}
\mathcal J_{\mathrm{MDPO}}(\theta\mid s)
=\frac1\eta\log Z_f(s)
-\frac1\eta\operatorname{KL}\!\left(\pi_\theta(\cdot\mid s)\middle\Vert\pi_k^+(\cdot\mid s)\right).
\label{eq:ch2-mdpo-kl-gap}
\end{equation}
MDPO 的 KL 方向与 PMD/BPO 中的 $K_{k,n}^\theta$ 不同：这里是
$\operatorname{KL}(\pi_\theta\Vert\pi_k^+)$，而中心化 log-ratio 的均值是
$\operatorname{KL}(\pi_k\Vert\pi_\theta)$。二者来自不同的积分测度，不能互换。

对式 \eqref{eq:ch2-mdpo-objective-bridge} 做 score-function 求导，得到
\begin{equation}
\nabla_\theta\mathcal J_{\mathrm{MDPO}}
=\mathbb E_{d_k,a\sim\pi_\theta}
\left[\operatorname{sg}\!\left(
f(s,a)-\frac1\eta\log\frac{\pi_\theta(a\mid s)}{\pi_k(a\mid s)}
\right)\nabla_\theta\log\pi_\theta(a\mid s)\right].
\label{eq:ch2-mdpo-gradient-bridge}
\end{equation}
推导中的 $-1/\eta$ 常数项乘以
$\mathbb E_{\pi_\theta}[\nabla\log\pi_\theta]=0$，所以最终式不需要引入
额外的 ratio 记号。

\begin{proposition}[平方回归的投影含义]
\label{prop:ch2-regression-projection}
若 student 函数类不能表示式 \eqref{eq:ch2-logratio-equation} 的右侧，
PMD-Part 的精确 inner 解是镜像方程在函数类诱导的 $L_2(d_k\pi_k)$ 几何下的
投影；若训练样本的历史访问测度改为 $m$，投影也相应变为 $L_2(m\pi_k)$。
因此 inner loss 为零只在 target 可表示且训练测度覆盖目标测度时才意味着精确
mirror 解。
\end{proposition}

\begin{proof}
平方损失是 Hilbert 空间中的正交投影问题。对任意可行扰动 $v$，最优残差
$\mathscr E^\star$ 满足
\[
\mathbb E_{m,\pi_k}[\mathscr E^\star v]=0.
\]
这正是镜像方程右侧到可行函数集合的正交投影条件。若 target 属于该集合，
投影残差为零；否则残差由函数类和测度共同决定。
\end{proof}

\section{四种 inner 的统一推导}
\label{sec:ch2-four-methods}

下面固定一个历史，令
\[
\ell_T(a)=\log\pi_T(a\mid s),\qquad
\ell_k(a)=\log\pi_k(a\mid s),\qquad
\ell_{k,n}(a)=\log\pi_{k,n}(a\mid s).
\]
训练时只对 $\ell_{k,n}$ 反向传播；teacher、anchor、目标常数和已经解析到
policy-gradient 权重中的 KL 都停止梯度。

\subsection{PMD-Part：回归完整 Gibbs log-ratio}
\label{sec:ch2-pmd-part}

PMD-Part 在 anchor 测度下最小化
\begin{equation}
\label{eq:ch2-part-loss}
\mathcal L_{\mathrm{Part}}(\theta)
=\frac12\mathbb E_{s\sim d_k,\,a\sim\pi_k}
\left[
\log\frac{\pi_\theta(a\mid s)}{\pi_k(a\mid s)}
-\eta f(s,a)+\log Z_f(s)
\right]^2.
\end{equation}
对 target 停止梯度，链式法则给出
\begin{equation}
\label{eq:ch2-part-gradient}
-\nabla_\theta\mathcal L_{\mathrm{Part}}
=\mathbb E_{d_k,\pi_k}
\left[\eta f(s,a)-\log Z_f(s)
-\log\frac{\pi_\theta(a\mid s)}{\pi_k(a\mid s)}\right]
\nabla_\theta\log\pi_\theta(a\mid s).
\end{equation}
这一步是普通的 log-ratio regression；它没有把 KL 视为常数，也没有使用 action-level importance ratio。

固定 history 后令 $q_{k,n}(a)=\Delta_{k,n}(s,a)-H_k(s,a)$。由于 anchor 测度与 target 都冻结，逐步求导得到
\begin{align}
\nabla_\theta\mathcal L_{\mathrm{Part}}(s)
&=\frac12\nabla_\theta\sum_a\pi_k(a\mid s)q_{k,n}(a)^2\\
&=\sum_a\pi_k(a\mid s)q_{k,n}(a)\nabla_\theta\Delta_{k,n}(s,a)\\
&=\E_{a\sim\pi_k}[q_{k,n}(a)\nabla_\theta\log\pi_{k,n}(a\mid s)\mid s].
\end{align}
若实际 rollout 来自行为测度 $\beta_k$，而目标仍是 $L_2(\pi_k)$ 投影，则每个 sampled 平方损失项乘以冻结权重 $\operatorname{sg}(\pi_k(a\mid s)/\beta_k(a\mid s))$。不加这个权重时，算法仍然有清晰含义，但求解的是 $L_2(\beta_k)$ 投影。

\begin{algorithm}[ht]
\caption{PMD-Part inner update\\{\small\textit{Semantics: 固定 anchor 下的完整 Gibbs log-ratio 回归}}}
\begin{algorithmic}[1]
\Require frozen $\pi_k$, teacher signal $f$, histories $\{s_i\}_{i=1}^B$, $\eta$, step size $\alpha$
\Ensure updated parameters $\theta$
\For{$i=1,\ldots,B$}
    \State sample $a_i\sim\pi_k(\cdot\mid s_i)$
    \State $\log Z_i\gets\operatorname{logsumexp}_{v\in\mathcal V}(\log\pi_k(v\mid s_i)+\eta f(s_i,v))$
    \State $H_i\gets\eta f(s_i,a_i)-\operatorname{stopgrad}(\log Z_i)$
    \State $U_i\gets\log\pi_\theta(a_i\mid s_i)-\operatorname{stopgrad}(\log\pi_k(a_i\mid s_i))$
    \State $\ell_i\gets\frac12\bigl(U_i-\operatorname{stopgrad}(H_i)\bigr)^2$
\EndFor
\State $\theta\gets\theta-\alpha\nabla_\theta(B^{-1}\sum_i\ell_i)$
\State \Return $\theta$
\end{algorithmic}
\end{algorithm}

\subsection{PMD-Mean：中心化 target 与归一化约束}
\label{sec:ch2-pmd-mean}

由定义可得
\[
\mathbb E_{a\sim\pi_k}[f(s,a)]=\bar f_k(s),
\qquad
\mathbb E_{a\sim\pi_k}[\widetilde f_k(s,a)]=0.
\]
PMD-Mean 用 $\eta\widetilde f_k$ 作为回归 target，并要求 student log-ratio 满足归一化约束
\begin{equation}
\label{eq:ch2-mean-loss}
\min_{U}\ \frac12\mathbb E_{\pi_k}\left[(U-\eta\widetilde f_k)^2\mid s\right]
\quad\text{s.t.}\quad
\mathbb E_{\pi_k}[e^{U}\mid s]=1.
\end{equation}

\begin{theorem}[PMD-Mean 的逐历史 Lambert-$W$ 投影]
\label{thm:ch2-mean-lambert}
在有限词表、$\pi_k(a\mid s)>0$ 且约束可行时，上述仅中心化 reward 的平方回归具有唯一全局最小点，其 KKT 解满足
\begin{equation}
\label{eq:ch2-mean-lambert-theorem}
U_a=G_a-W\!\left(\lambda(s)e^{G_a}\right),\qquad
G_a=\eta\widetilde f_k(s,a),
\end{equation}
其中 $\lambda(s)\geq0$ 由 $\sum_a\pi_k(a\mid s)e^{U_a}=1$ 唯一确定。
\end{theorem}

\begin{proof}
以下给出固定历史 $s$ 的 KKT 推导。令词表上的函数值为 $U_a$ 和 $G_a=\eta\widetilde f_k(s,a)$，拉格朗日函数为
\[
\mathscr L(U,\lambda)
=\frac12\sum_a\pi_k(a\mid s)(U_a-G_a)^2
+\lambda\left(\sum_a\pi_k(a\mid s)e^{U_a}-1\right).
\]
一阶条件为
\[
U_a-G_a+\lambda e^{U_a}=0.
\]
令 $V_a=G_a-U_a$，则 $V_a e^{V_a}=\lambda e^{G_a}$，所以
\begin{equation}
\label{eq:ch2-mean-lambert}
U_a=G_a-W\!\left(\lambda e^{G_a}\right),
\end{equation}
其中 $\lambda$ 由归一化约束唯一确定。为说明唯一性，定义
\[
\Phi(\lambda)=\sum_a\pi_k(a\mid s)\exp\!\left(G_a-W(\lambda e^{G_a})\right),\qquad \lambda\geq0.
\]
由于 $\mathbb E_{\pi_k}[G\mid s]=0$，Jensen 不等式给出
$\Phi(0)=\mathbb E_{\pi_k}[e^G\mid s]\ge1$。Lambert--$W$ 的主分支在非负轴上严格递增，所以 $\Phi$ 连续且严格递减，且 $\Phi(\lambda)\to0$ 当 $\lambda\to\infty$。因此归一化方程有唯一非负根；非恒定 $G$ 对应严格正根，$G\equiv0$ 时根为零。

对于该根 $\lambda\ge0$，拉格朗日函数的 Hessian 是
\[
\nabla_U^2\mathscr L
=\operatorname{diag}\!\left(\pi_k(a\mid s)(1+\lambda e^{U_a})\right)\succ0.
\]
因此构造出的 $U$ 唯一最小化这个拉格朗日函数。所有满足归一化约束的 $U$ 都有 $\mathscr L(U,\lambda)=\mathcal L_{\mathrm{Mean}}(U)$，构造的最小点又满足约束，故它是原受约束问题的唯一全局最小点。这一论证不要求指数归一化等式的可行集是凸集。
\end{proof}

约束 $\sum_a\pi_\theta(a\mid s)=1$ 正是 log-ratio 坐标中的指数归一化条件。精确 Gibbs 解为
\[
U_a^{\mathrm{Gibbs}}=G_a-\delta_f(s).
\]
将它代入 PMD-Mean 的 KKT 条件，要求
\[
-\delta_f(s)+\lambda e^{G_a-\delta_f(s)}=0
\]
对所有 action 同时成立。非恒定 $G$ 时 $\delta_f(s)>0$，且 $e^{G_a-\delta_f(s)}$ 随 action 改变，因此不存在这样的单一 $\lambda$。PMD-Mean 的 Lambert-$W$ 修正是 action-dependent，原始 Gibbs 的配分函数修正则是 action-independent；二者只有在信号恒定时重合。

本节损失的乘子 $\lambda$ 对应原论文乘子除以 $\tau^2$，其中 $\tau=1/\eta$。对 KKT 条件按 anchor 求均值，由 $\mathbb E_{\pi_k}[G]=0$ 和 $\mathbb E_{\pi_k}[e^U]=1$ 得到
\[
\lambda=-\mathbb E_{\pi_k}[U]
=\operatorname{KL}(\pi_k\Vert\pi_{\mathrm{Mean}}^*).
\]
固定这个由解确定的 $\lambda$，同一分布也唯一最大化
\begin{equation}
\mathbb E_p[f]-\frac1\eta\operatorname{KL}(p\Vert\pi_k)
-\frac{\lambda}{2\eta}\chi^2(p\Vert\pi_k),
\qquad
\chi^2(p\Vert\pi_k)=\mathbb E_{\pi_k}\left[\left(\frac p{\pi_k}-1\right)^2\right].
\label{eq:ch2-mean-mixed-objective}
\end{equation}
为验证这一等价，令 $\xi$ 为概率归一化乘子。混合目标的逐 action 一阶条件为
\[
f_a-\frac1\eta(U_a+1)-\frac\lambda\eta(e^{U_a}-1)+\xi=0.
\]
在 PMD-Mean 解处有 $U_a+\lambda e^{U_a}=\eta(f_a-\bar f_k)$；取
$\xi=(1-\lambda)/\eta-\bar f_k$ 即满足上式。混合目标中的负 KL 严格凹，负 $\chi^2$ 凹，因此这个驻点是唯一全局最大点。这里的等价指总体最优分布相同，不表示两个训练目标在任意参数点上的梯度相同。

实际样本只需估计冻结的 reward center。给定同一 history 的 $K\ge2$ 个独立 anchor actions，使用 leave-one-out 均值
\[
\widehat{\bar f}_{-j}=\frac1{K-1}\sum_{\ell\ne j}f(s,a_\ell).
\]
student log-ratio $U_\theta(s,a_j)$ 保持原值，不计算或减去 $\bar U_\theta$。因为该 center 不依赖 student，leave-one-out reward center 的有限样本平方损失与总体损失只相差一个参数无关的条件方差项，其梯度无偏。
若 $K=1$，需要外部同 prefix 的 reward center；使用不准确的冻结 baseline 会改变回归目标，必须单独记录该误差。

\begin{algorithm}[ht]
\caption{PMD-Mean inner update\\{\small\textit{Semantics: 中心化 target 在指数归一化集合上的投影}}}
\begin{algorithmic}[1]
\Require frozen $\pi_k$, signal $f$, histories $\{s_i\}$, $K\ge2$, $\eta$, step size $\alpha$
\Ensure updated parameters $\theta$
\For{each history $s_i$}
    \For{$j=1,\ldots,K$}
        \State sample $a_{ij}\sim\pi_k(\cdot\mid s_i)$
        \State $U_{ij}\gets\log\pi_\theta(a_{ij}\mid s_i)-\operatorname{stopgrad}(\log\pi_k(a_{ij}\mid s_i))$
        \State $F_{ij}\gets f(s_i,a_{ij})$
    \EndFor
    \For{$j=1,\ldots,K$}
        \State $\bar F_{-j}\gets(K-1)^{-1}\sum_{\ell\ne j}F_{i\ell}$
        \State $\ell_{ij}\gets\frac12[U_{ij}-\eta\operatorname{stopgrad}(F_{ij}-\bar F_{-j})]^2$
    \EndFor
\EndFor
\State $\theta\gets\theta-\alpha\nabla_\theta(BK)^{-1}\sum_{i,j}\ell_{ij}$
\State \Return $\theta$
\end{algorithmic}
\end{algorithm}


\subsection{BPO：中心化最优性 residual}
\label{sec:ch2-bpo}

BPO 不显式回归 $\log Z_f$，而将同一个最优性方程的 action-independent 部分消去。定义
\begin{equation}
\label{eq:ch2-bpo-residual}
D_{k,n}(s,a)
=\eta f(s,a)-\Delta_{k,n}(s,a)
-\mathbb E_{v\sim\pi_k}[\eta f(s,v)-\Delta_{k,n}(s,v)\mid s].
\end{equation}
直接展开可得
\[
D_{k,n}(s,a)
=\eta\widetilde f_k(s,a)-\Delta_{k,n}(s,a)
+\mathbb E_{v\sim\pi_k}[\Delta_{k,n}(s,v)\mid s].
\]
未线性化 BPO 的损失为
\begin{equation}
\label{eq:ch2-bpo-loss}
\mathcal L_{\mathrm{BPO}}(\theta)
=\frac1{2\eta}\mathbb E_{d_k,\pi_k}[D_{k,n}(s,a)^2].
\end{equation}
由于
\[
\nabla_\theta\mathbb E_{v\sim\pi_k}[\Delta_{k,n}(s,v)]
=\mathbb E_{v\sim\pi_k}[\nabla_\theta\log\pi_{k,n}(v\mid s)],
\]
所以
\begin{equation}
\label{eq:ch2-bpo-gradient}
-\eta\nabla_\theta\mathcal L_{\mathrm{BPO}}
=\mathbb E_{d_k,\pi_k}\left[D_{k,n}(s,a)
\left(\nabla_\theta\log\pi_{k,n}(a\mid s)
 -\mathbb E_{v\sim\pi_k}[\nabla_\theta\log\pi_{k,n}(v\mid s)\mid s]\right)\right].
\end{equation}
BPO 的 score centering 来自对当前 student--anchor KL 的精确求导；它不是因为先对 reward 做了一个近似。

令 $G_{k,n}(s)=\E_{\pi_k}[\eta f(s,v)-\Delta_{k,n}(s,v)\mid s]$，则
\[
\nabla_\theta D_{k,n}(s,a)=-\nabla_\theta\Delta_{k,n}(s,a)+\nabla_\theta\E_{\pi_k}[\Delta_{k,n}(s,v)\mid s]
=-\nabla_\theta\log\pi_{k,n}(a\mid s)+\bar g_{k,n}(s).
\]
所以 centered score 是对 center 本身求导的结果，而不是把 reward 任意减去 baseline。若用独立 action $a_i$ 与 $K$ 个独立 auxiliary actions 估计 center，令 $z=\eta f-U_\theta$，则
\[
\mathbb E\left[\left(z(a_i)-\frac1K\sum_{\ell=1}^Kz(b_{i\ell})\right)^2\right]
=\left(1+\frac1K\right)\operatorname{Var}_{\pi_k}(z).
\]
因此直接对这个 sampled 平方损失反向传播时，需乘以 $K/(K+1)$ 才有原始 BPO 损失的无偏梯度。leave-one-out 配置使用 $K-1$ 个 center 样本，对应系数为 $(K-1)/K$。仅令 residual 样本与 center 样本独立，并不能消除 residual 与 center 导数乘积中的协方差。

\begin{algorithm}[ht]
\caption{BPO inner update\\{\small\textit{Semantics: 对中心化最优性 residual 求平方损失}}}
\begin{algorithmic}[1]
\Require frozen $\pi_k$, signal $f$, histories $\{s_i\}$, center size $K$, $\eta$, step size $\alpha$
\Ensure updated parameters $\theta$
\For{each history $s_i$}
    \State sample $a_i\sim\pi_k(\cdot\mid s_i)$ and $b_{i1},\ldots,b_{iK}\sim\pi_k(\cdot\mid s_i)$ independently
    \State $z_i\gets\eta f(s_i,a_i)-\log\pi_\theta(a_i\mid s_i)+\operatorname{stopgrad}(\log\pi_k(a_i\mid s_i))$
    \State $\bar z_i\gets K^{-1}\sum_{\ell=1}^K[\eta f(s_i,b_{i\ell})-\log\pi_\theta(b_{i\ell}\mid s_i)+\operatorname{stopgrad}(\log\pi_k(b_{i\ell}\mid s_i))]$
    \State $d_i\gets z_i-\bar z_i$, $\ell_i\gets\frac{K}{K+1}\frac{d_i^2}{2\eta}$
\EndFor
\State $\theta\gets\theta-\alpha\nabla_\theta B^{-1}\sum_i\ell_i$
\State \Return $\theta$
\end{algorithmic}
\end{algorithm}

若从行为测度 $\beta$ 采样，不能先把 $z$ 乘以 importance weight 再平方；这会把二阶矩的权重变成平方。可用两个独立样本 $a,b\sim\beta(\cdot\mid s)$，令 $w(v)=\operatorname{sg}(\pi_k(v\mid s)/\beta(v\mid s))$，使用
\[
\widehat{\mathcal L}_{\mathrm{BPO}}(s)
=\frac{w(a)w(b)}{4\eta}[z(a)-z(b)]^2.
\]
因为 $\mathbb E_{a,b\sim\pi_k}[(z(a)-z(b))^2]=2\operatorname{Var}_{\pi_k}(z)$，上述估计器的期望恰为原始 BPO 损失，且固定权重下 autograd 给出无偏梯度。

\subsection{MDPO：直接优化 mirror objective}
\label{sec:ch2-mdpo-generic}

MDPO 直接最大化式 \eqref{eq:ch2-mirror} 的参数化版本：
\begin{equation}
\label{eq:ch2-mdpo-generic-objective}
\mathcal J_{\mathrm{MDPO}}(\theta)
=\mathbb E_{a\sim\pi_\theta}[f(s,a)]
-\frac1\eta\operatorname{KL}\!\left(\pi_\theta(\cdot\mid s)\middle\Vert\pi_k(\cdot\mid s)\right).
\end{equation}
完整词表时 KL 是
\begin{equation}
\label{eq:ch2-mdpo-full-kl}
\operatorname{KL}(\pi_\theta\Vert\pi_k)
=\sum_a\pi_\theta(a\mid s)
\left[\log\pi_\theta(a\mid s)-\log\pi_k(a\mid s)\right].
\end{equation}
若 action 从 anchor 采样，则只需把 student 期望改写为显式的
指数权重，不能把它另记成一个容易与 reward 混淆的 $R_\theta$：
\begin{equation}
\label{eq:ch2-mdpo-sample-integrand}
\widehat{\mathcal J}_{\mathrm{MDPO}}
=\mathbb E_{a\sim\pi_k}\left[
\exp\!\left(\log\frac{\pi_\theta(a\mid s)}{\pi_k(a\mid s)}\right)
\left(f(s,a)-\frac1\eta\log\frac{\pi_\theta(a\mid s)}{\pi_k(a\mid s)}\right)
\right].
\end{equation}
其 ascent direction 为
\begin{equation}
\label{eq:ch2-mdpo-generic-gradient}
\nabla_\theta\mathcal J_{\mathrm{MDPO}}
=\mathbb E_{\pi_k}\left[
\exp\!\left(\log\frac{\pi_\theta(a\mid s)}{\pi_k(a\mid s)}\right)
\left(f(s,a)-\frac1\eta\log\frac{\pi_\theta(a\mid s)}{\pi_k(a\mid s)}\right)
\nabla_\theta\log\pi_\theta(a\mid s)\right].
\end{equation}
额外的 $-1/\eta$ 项乘以 score 后，在完整 action 求和中因
$\mathbb E_{\pi_\theta}[\nabla\log\pi_\theta]=0$ 消失。MDPO 因而直接优化
mirror objective，其他三种方法优化的是由同一最优性方程构造的 regression objective。

MDPO 的自然梯度只改变更新几何，不改变上面的 token target。令 $g=\nabla_\theta\mathcal J_{\mathrm{MDPO}}$，并令
\[
F_\theta=\E_{a\sim\pi_\theta}[g_\theta(a)g_\theta(a)^\top],\qquad
g_\theta(a)=\nabla_\theta\log\pi_\theta(a\mid s).
\]
若从 anchor 采样，用
$\operatorname{sg}\!\left(\exp(\log\pi_\theta-\log\pi_k)\right)$ 进行测度校正。
由于 categorical Fisher 的零空间是 logits 的常数方向，$F_\theta^\dagger g$
位于 simplex 切空间；神经网络参数化下则需用阻尼逆
$(F+\lambda I)^{-1}$ 或 KL trust region，不能把自然步长与 mirror 温度 $\eta$
混用。

\begin{algorithm}[ht]
\caption{MDPO natural-gradient inner update\\{\small\textit{Semantics: 直接优化固定 history 的 mirror objective}}}
\begin{algorithmic}[1]
\Require frozen $\pi_k$, signal $f$, histories $\{s_i\}$, $\eta$, natural step $\alpha$
\Ensure updated parameters $\theta$
\For{each history $s_i$}
    \State sample $a_i\sim\pi_k(\cdot\mid s_i)$
    \State $w_i\gets\exp(\log\pi_\theta(a_i\mid s_i)-\operatorname{stopgrad}(\log\pi_k(a_i\mid s_i)))$
    \State $j_i\gets w_i\left[f(s_i,a_i)-\eta^{-1}\log w_i\right]$ \Comment{保留 $w_i$ 对 student 的梯度}
\EndFor
\State $g\gets\nabla_\theta(B^{-1}\sum_i j_i)$
\State $g_i^{\mathrm{score}}\gets\nabla_\theta\log\pi_\theta(a_i\mid s_i)$
\State $F\gets B^{-1}\sum_i\operatorname{stopgrad}(w_i)g_i^{\mathrm{score}}(g_i^{\mathrm{score}})^\top$
\State $\theta\gets\theta+\alpha(F+\lambda I)^\dagger g$
\State \Return $\theta$
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Gibbs 解与 PMD-Mean 的 Lambert-$W$ 解]
\label{thm:ch2-inner-common-root}
在 tabular policy、完整 action 支持和精确词表期望下，PMD-Part、未线性化 BPO 和 MDPO 的总体最优解是式~\eqref{eq:ch2-optimal-policy} 的 Gibbs 分布。PMD-Mean 的总体最优解是定理~\ref{thm:ch2-mean-lambert} 的 Lambert-$W$ 分布；非恒定信号下与 Gibbs 分布不同。参数化 student 或有限样本还会引入函数类投影、采样及优化误差。
\end{theorem}

\begin{proof}
PMD-Part 在零损失处满足 $U_\theta=\eta f-\log Z_f$，得到 Gibbs 分布。BPO 的零 residual 只要求 $U_\theta=\eta f-C(s)$；代入 $\mathbb E_{\pi_k}[e^{U_\theta}]=1$ 得 $C(s)=\log Z_f(s)$。MDPO 由式~\eqref{eq:ch2-kl-decomp} 得到唯一最大点。PMD-Mean 的受约束全局最小点及与 Gibbs 解的分离已在定理~\ref{thm:ch2-mean-lambert} 后证明。
\end{proof}

\begin{proposition}[PMD-Mean 与 BPO 的损失和梯度差异]
\label{prop:ch2-mean-bpo-gap}
固定历史 $s$，记 $U=\log(\pi_\theta/\pi_k)$、$G=\eta(f-\bar f_k)$、
$K^\theta=\operatorname{KL}(\pi_k\Vert\pi_\theta)$。则
\begin{align}
\mathcal L_{\mathrm{Mean}}(s)
&=\eta\mathcal L_{\mathrm{BPO}}(s)+\frac12(K^\theta(s))^2,\\
\nabla_\theta\mathcal L_{\mathrm{Mean}}(s)
&=\eta\nabla_\theta\mathcal L_{\mathrm{BPO}}(s)
+K^\theta(s)\nabla_\theta K^\theta(s).
\label{eq:ch2-mean-bpo-gap}
\end{align}
\end{proposition}

\begin{proof}
因为 $\mathbb E_{\pi_k}[G]=0$，$\mathbb E_{\pi_k}[U-G]=-K^\theta$，且
$D=G-U-K^\theta$，故 $U-G=-D-K^\theta$。平方展开并使用 $\mathbb E_{\pi_k}[D]=0$ 得
\[
\frac12\mathbb E_{\pi_k}[(U-G)^2]
=\frac12\mathbb E_{\pi_k}[D^2]+\frac12(K^\theta)^2.
\]
代入 BPO 的 $1/(2\eta)$ 系数，再求导即得两式。写成同尺度的下降方向，得到
\[
-\nabla\mathcal L_{\mathrm{Mean}}
=-\eta\nabla\mathcal L_{\mathrm{BPO}}
+K^\theta\mathbb E_{\pi_k}[\nabla\log\pi_\theta].
\]
在 inner 初始化 $\pi_\theta=\pi_k$ 时 $K^\theta=0$，两个方向相同；离开 anchor 后差异项一般保留。tabular softmax logits 中
$\mathbb E_{\pi_k}[\nabla_z\log\pi_\theta]=\pi_k-\pi_\theta$，额外方向为
$K^\theta(\pi_k-\pi_\theta)$。
\end{proof}

\begin{proposition}[未线性化 BPO 与 PMD-Part 的梯度差异]
\label{prop:ch2-bpo-part-gradient-gap}
固定一个历史 $s$，令
\begin{equation}
\label{eq:ch2-bpo-part-qc}
q(s,a)=\eta f(s,a)-\log Z_f(s)-U_\theta(s,a),
\qquad
c(s)=\log Z_f(s)+\mathbb E_{\pi_k}[U_\theta(s,\cdot)\mid s]-\eta\mathbb E_{\pi_k}[f(s,\cdot)\mid s].
\end{equation}
则未线性化 BPO residual 满足 $D_\theta=q+c$，并且
\begin{equation}
\label{eq:ch2-bpo-part-gradient-gap}
-\eta\nabla_\theta\mathcal L_{\mathrm{BPO}}
=-\nabla_\theta\mathcal L_{\mathrm{Part}}
+c(s)\,\mathbb E_{a\sim\pi_k}[\nabla_\theta\log\pi_\theta(a\mid s)\mid s].
\end{equation}
因此两者在精确镜像解处都有零梯度，但在一般参数点不相同。它们只有在
$c(s)\,\mathbb E_{\pi_k}[\nabla_\theta\log\pi_\theta]=0$
时才给出相同的方向。
\end{proposition}

\begin{proof}
从 PMD-Part 的最优性残差出发，先把 BPO 的中心化式逐项展开：
\begin{align*}
D_\theta(s,a)
&=\eta f(s,a)-U_\theta(s,a)
 -\mathbb E_{\pi_k}[\eta f(s,\cdot)-U_\theta(s,\cdot)\mid s]\\
&=\eta f(s,a)-\log Z_f(s)-U_\theta(s,a)
 +\log Z_f(s)+\mathbb E_{\pi_k}[U_\theta(s,\cdot)\mid s]\\
&=q(s,a)+c(s).
\end{align*}
由定义，$\mathbb E_{\pi_k}[D_\theta\mid s]=0$。BPO 的精确梯度为
\begin{align*}
-\eta\nabla_\theta\mathcal L_{\mathrm{BPO}}
&=\mathbb E_{\pi_k}\left[D_\theta
 \left(g_\theta-\bar g_\theta\right)\mid s\right],\\
 g_\theta(a)&=\nabla_\theta\log\pi_\theta(a\mid s),
 \qquad \bar g_\theta=\mathbb E_{\pi_k}[g_\theta\mid s].
\end{align*}
代入 $D_\theta=q+c$，并使用 $\mathbb E_{\pi_k}[D_\theta\mid s]=0$，可得
\begin{align*}
\mathbb E_{\pi_k}[D_\theta(g_\theta-\bar g_\theta)\mid s]
&=\mathbb E_{\pi_k}[(q+c)g_\theta\mid s]
 -\mathbb E_{\pi_k}[D_\theta\mid s]\bar g_\theta\\
&=\mathbb E_{\pi_k}[qg_\theta\mid s]+c(s)\bar g_\theta.
\end{align*}
而 PMD-Part 的梯度式为
$-\nabla_\theta\mathcal L_{\mathrm{Part}}=\mathbb E_{\pi_k}[qg_\theta\mid s]$，故式
\eqref{eq:ch2-bpo-part-gradient-gap}
成立。若 $q=0$，则由归一化约束有 $c=0$，两边同时为零；离开该点时 $c$ 通常非零，差异项保留。
\end{proof}

这个命题说明“都在 token level”只保证它们面对同一个局部镜像根，不保证优化轨迹相同。PMD-Part 回归完整的 action-dependent log-ratio，并把 $\log Z_f$ 当作固定 target；BPO 先消去 action-independent 常数，再对当前 student 的归一化项求导，因此多出 centered score。若使用的参数化使
$\mathbb E_{\pi_k}[\nabla_\theta\log\pi_\theta]=0$
（例如采样测度与当前 student 一致，或在特定 tabular 点上），差异项才会消失。
